# 端口DNS

> 端口检测(nc/telnet)、DNS 诊断(dig/nslookup)、端口占用排查(ss/lsof)、Python 端口扫描。

## 一、端口检测

### 1.1 nc / ncat——端口连通性测试

```bash
#!/bin/bash

# ===== TCP 端口检测——最常用 =====
nc -zv 192.168.1.10 80                       # -z=不发送数据 -v=详细输出
nc -zv -w 3 192.168.1.10 22                  # -w=超时3秒

# 端口范围扫描
nc -zv 192.168.1.10 80-100                   # 扫描 80~100 端口

# ===== UDP 端口 =====
nc -zuv 192.168.1.10 53

# ===== 发送数据测试 =====
echo "GET / HTTP/1.0\r\n\r\n" | nc 192.168.1.10 80  # 测试 HTTP
echo "PING" | nc -w 1 192.168.1.10 6379              # 测试 Redis

# ===== telnet 替代（传统但被 nc 取代）=====
# timeout 3 bash -c "echo > /dev/tcp/192.168.1.10/80" && echo "通" || echo "不通"
```

### 1.2 lsof / ss——查看端口被谁占用

```bash
#!/bin/bash

# ===== lsof =====
lsof -i :80                                    # 查看 80 端口被哪个进程占用
lsof -i :80-100                                # 80~100 端口
lsof -i tcp:80                                 # 只看 TCP
lsof -i @192.168.1.10                          # 查看连接到某 IP 的
lsof -i -P -n                                  # 全部网络连接（不解析端口/主机名）

# ===== ss =====
ss -tlnp | grep :80                            # 列出 TCP 监听中 80 端口
ss -tlnp 'sport = :80'                         # ss 原生过滤

# ===== 找出谁在监听 =====
p80=$(ss -tlnp 'sport = :80' | awk 'NR>1{print $NF}' | grep -oP '(?<=pid=)\d+')
[[ -n "$p80" ]] && ps -o pid,user,comm -p "$p80"
```

### 1.3 Python——端口扫描器

```python
#!/usr/bin/env python3
"""快速 TCP 端口扫描——并发检测"""
import socket, sys, concurrent.futures

def check_port(ip, port, timeout=2):
    """检测单个端口是否开放"""
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        s.settimeout(timeout)
        return port if s.connect_ex((ip, port)) == 0 else None

def scan_ports(ip, ports):
    """并发扫描端口列表"""
    open_ports = []
    with concurrent.futures.ThreadPoolExecutor(max_workers=100) as ex:
        futures = {ex.submit(check_port, ip, p): p for p in ports}
        for fut in concurrent.futures.as_completed(futures):
            result = fut.result()
            if result: open_ports.append(result)
    return sorted(open_ports)

# ---- 命令行工具 ----
if __name__ == '__main__':
    if len(sys.argv) < 2:
        print(f"用法: {sys.argv[0]} <IP> [起始端口] [结束端口]")
        sys.exit(1)

    ip = sys.argv[1]
    start = int(sys.argv[2]) if len(sys.argv) > 2 else 1
    end   = int(sys.argv[3]) if len(sys.argv) > 3 else 1024

    print(f"扫描 {ip} 端口 {start}-{end} ...")
    open_ports = scan_ports(ip, range(start, end + 1))

    if open_ports:
        print(f"开放端口: {', '.join(map(str, open_ports))}")
    else:
        print("未发现开放端口")
```

## 二、DNS 诊断

### 2.1 dig——DNS 瑞士军刀

```bash
#!/bin/bash

# ===== 基础查询 =====
dig example.com                                # 完整信息
dig example.com +short                         # 只输出 IP
dig example.com A                              # 查询 A 记录
dig example.com AAAA                           # IPv6
dig example.com MX                             # 邮件服务器
dig example.com NS                             # 域名服务器
dig example.com TXT                            # TXT 记录（SPF/DKIM）

# ===== 指定 DNS 服务器 =====
dig @8.8.8.8 example.com                      # Google DNS
dig @1.1.1.1 example.com                      # Cloudflare

# ===== 追踪解析过程 =====
dig example.com +trace                         # 从根服务器逐级追踪

# ===== 反向解析 IP → 域名 =====
dig -x 8.8.8.8

# ===== 查看解析耗时 =====
dig example.com | grep "Query time"
```

### 2.2 nslookup——传统交互式查询

```bash
#!/bin/bash

nslookup example.com
nslookup example.com 8.8.8.8                   # 指定 DNS

# 交互模式
# > server 8.8.8.8        切换 DNS
# > set type=MX            查询 MX
# > example.com
# > exit
```

### 2.3 Python DNS 查询

```python
#!/usr/bin/env python3
"""Python DNS 查询——跨平台无依赖"""
import socket

def dns_lookup(domain, record_type='A'):
    """解析域名——纯 Python，无需额外安装"""
    try:
        if record_type == 'A':
            return socket.gethostbyname(domain)
        elif record_type == 'ALL':
            return socket.getaddrinfo(domain, None)
    except socket.gaierror as e:
        return f"解析失败: {e}"

print(dns_lookup('baidu.com'))
```

## 三、实战：网络环境检测脚本

```bash
#!/bin/bash
# 综合端口 + DNS 检测

echo "===== 网络环境检测 ====="

# 1. DNS 测试
echo "--- DNS 解析 ---"
for domain in baidu.com google.com github.com; do
    if host "$domain" > /dev/null 2>&1; then
        echo "✅ $domain ($(dig +short "$domain" | head -1))"
    else
        echo "❌ $domain 解析失败"
    fi
done

# 2. 端口测试
echo ""
echo "--- 端口连通性 ---"
declare -A TARGETS=(
    ["www.baidu.com:443"]="外网HTTPS"
    ["8.8.8.8:53"]="DNS"
    ["localhost:22"]="SSH"
)
for target in "${!TARGETS[@]}"; do
    host="${target%:*}"
    port="${target#*:}"
    if timeout 2 bash -c "echo > /dev/tcp/$host/$port" 2>/dev/null; then
        echo "✅ ${TARGETS[$target]} ($host:$port)"
    else
        echo "❌ ${TARGETS[$target]} ($host:$port)"
    fi
done
```

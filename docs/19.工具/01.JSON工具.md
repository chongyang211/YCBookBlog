<script>
export default {
  components: {
    JsonTool: {
      template: `
        <div class="json-tool-container">
          <div class="json-panel">
            <div class="panel-header">
              <span class="panel-title">输入 JSON</span>
              <div class="btn-group">
                <button class="btn" @click="format">格式化</button>
                <button class="btn" @click="compress">压缩</button>
                <button class="btn btn-primary" @click="validate">校验</button>
                <button class="btn" @click="clear">清空</button>
              </div>
            </div>
            <textarea
              v-model="input"
              placeholder='粘贴 JSON 到这里，例如：{"name":"编程进阶","version":1}'
              class="json-textarea"
              spellcheck="false"
            ></textarea>
          </div>

          <div class="json-panel" v-if="output || errorMsg">
            <div class="panel-header">
              <span class="panel-title" :style="{ color: errorMsg ? '#e74c3c' : '#27ae60' }">
                {{ errorMsg ? '❌ 错误' : '✅ 输出' }}
              </span>
              <button class="btn" @click="copyResult" v-if="!errorMsg && output">复制</button>
            </div>
            <pre class="json-output" :class="{ error: errorMsg }">{{ errorMsg || output }}</pre>
          </div>

          <div class="json-panel" v-if="info">
            <div class="panel-header">
              <span class="panel-title">📊 统计信息</span>
            </div>
            <div class="info-grid">
              <div class="info-item"><span class="info-label">大小</span><span>{{ info.size }}</span></div>
              <div class="info-item"><span class="info-label">行数</span><span>{{ info.lines }}</span></div>
              <div class="info-item"><span class="info-label">顶级类型</span><span>{{ info.type }}</span></div>
              <div class="info-item"><span class="info-label">键总数</span><span>{{ info.keys }}</span></div>
              <div class="info-item"><span class="info-label">最大嵌套</span><span>{{ info.depth }}</span></div>
            </div>
          </div>
        </div>
      `,
      data() {
        return {
          input: '',
          output: '',
          errorMsg: '',
          info: null,
        }
      },
      methods: {
        getInfo(obj) {
          const jsonStr = JSON.stringify(obj, null, 2)
          const lines = jsonStr.split('\n').length
          const size = new Blob([jsonStr]).size
          const sizeStr = size < 1024 ? size + ' B' : size < 1048576 ? (size / 1024).toFixed(1) + ' KB' : (size / 1048576).toFixed(1) + ' MB'

          let keys = 0, maxDepth = 0
          const countKeys = (o, depth) => {
            if (depth > maxDepth) maxDepth = depth
            if (Array.isArray(o)) {
              o.forEach(item => { if (typeof item === 'object' && item !== null) countKeys(item, depth + 1) })
            } else if (typeof o === 'object' && o !== null) {
              keys += Object.keys(o).length
              Object.values(o).forEach(v => { if (typeof v === 'object' && v !== null) countKeys(v, depth + 1) })
            }
          }
          countKeys(obj, 1)

          const type = Array.isArray(obj) ? 'Array' : typeof obj === 'object' ? 'Object' : typeof obj
          return { size: sizeStr, lines, type, keys, depth: maxDepth }
        },

        format() {
          this.errorMsg = ''
          this.info = null
          try {
            const obj = JSON.parse(this.input)
            this.output = JSON.stringify(obj, null, 2)
            this.info = this.getInfo(obj)
          } catch (e) {
            this.errorMsg = 'JSON 格式错误：' + e.message
            this.output = ''
          }
        },

        compress() {
          this.errorMsg = ''
          this.info = null
          try {
            const obj = JSON.parse(this.input)
            this.output = JSON.stringify(obj)
            this.info = this.getInfo(obj)
          } catch (e) {
            this.errorMsg = 'JSON 格式错误：' + e.message
            this.output = ''
          }
        },

        validate() {
          this.info = null
          try {
            const obj = JSON.parse(this.input)
            this.output = ''
            const type = Array.isArray(obj) ? 'Array' : 'Object'
            this.errorMsg = ''
            this.info = this.getInfo(obj)
            alert('✅ JSON 校验通过！\\n类型：' + type)
          } catch (e) {
            this.errorMsg = '❌ JSON 校验失败：' + e.message
            this.output = ''
          }
        },

        clear() {
          this.input = ''
          this.output = ''
          this.errorMsg = ''
          this.info = null
        },

        copyResult() {
          if (this.output) {
            navigator.clipboard.writeText(this.output).then(() => {
              alert('已复制到剪贴板')
            })
          }
        }
      }
    }
  }
}
</script>

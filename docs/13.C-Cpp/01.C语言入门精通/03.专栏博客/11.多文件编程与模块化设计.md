## 一、引言：为什么需要多文件编程

### 1.1 单文件的困境

```c
// 所有代码写在一个main.c中的问题：
// 1. 文件太长，难以导航和维护（几千行+）
// 2. 多人协作容易冲突
// 3. 修改任何一行都要重新编译整个文件
// 4. 代码无法复用（复制粘贴不是复用）
// 5. 无法单独测试各个功能模块
```

### 1.2 多文件编程的优势

```
单文件:  main.c (10000行)
    ↓
多文件:  main.c (200行) + network.c + database.c + utils.c + ...

优势：
1. 逻辑分离：每个文件负责一个功能模块
2. 增量编译：只重编译修改过的文件
3. 代码复用：模块可被多个项目使用
4. 团队协作：不同人负责不同模块
5. 信息隐藏：模块内部细节不暴露
```

### 1.3 一个典型的多文件项目结构

```
project/
├── include/           # 公开头文件
│   ├── list.h
│   ├── hashmap.h
│   └── logger.h
├── src/               # 源文件
│   ├── main.c
│   ├── list.c
│   ├── hashmap.c
│   └── logger.c
├── test/              # 测试文件
│   ├── test_list.c
│   └── test_hashmap.c
├── lib/               # 第三方库
├── Makefile
└── README.md
```

## 二、编译单元与翻译单元

### 2.1 翻译单元（Translation Unit）

```
翻译单元 = 一个.c源文件 + 它包含的所有头文件内容

预处理后，每个.c文件成为一个独立的翻译单元。
编译器独立编译每个翻译单元，生成对应的.o文件。

main.c + (stdio.h + string.h + mylib.h) → main.i → main.o
list.c + (stdlib.h + list.h)            → list.i → list.o
```

### 2.2 独立编译的含义

```c
// a.c
int global_var = 42;
void func_a() { /* ... */ }

// b.c
extern int global_var;    // 声明a.c中的变量
extern void func_a();     // 声明a.c中的函数
void func_b() {
    func_a();             // 编译b.c时，编译器不知道func_a的实现
    printf("%d\n", global_var);  // 也不知道global_var在哪里定义
}

// 编译器只检查声明的正确性，实际的连接在链接阶段完成
// gcc -c a.c → a.o  (func_a是已定义符号)
// gcc -c b.c → b.o  (func_a是未定义符号)
// gcc a.o b.o → a.out (链接器将b.o中的func_a引用解析到a.o中的定义)
```

## 三、头文件的设计原则

### 3.1 头文件中应该放什么

```c
// mymodule.h

#ifndef MYMODULE_H
#define MYMODULE_H

// 1. 包含保护（必须）
// 2. 必要的#include
#include <stddef.h>

// 3. 宏定义
#define MODULE_VERSION "1.0"
#define MAX_ITEMS 100

// 4. 类型定义
typedef struct {
    int id;
    char name[64];
} Item;

typedef enum {
    STATUS_OK,
    STATUS_ERROR,
    STATUS_NOT_FOUND
} Status;

// 5. 函数声明（接口）
Status module_init(void);
Status module_add_item(const Item *item);
Item *module_find_item(int id);
void module_cleanup(void);

// 6. 外部变量声明
extern int module_item_count;

// 7. static inline函数
static inline int module_is_empty(void) {
    return module_item_count == 0;
}

#endif // MYMODULE_H
```

### 3.2 头文件中不应该放什么

```c
// 错误示例

// ❌ 变量定义（会导致多重定义）
int global_var = 42;  // 每个包含此头文件的.c都会定义一次

// ❌ 函数定义（除非是static inline）
void helper() {       // 多重定义错误
    printf("helper\n");
}

// ❌ 不相关的#include
#include <mysql.h>    // 不应该把实现细节暴露给用户
#include <pthread.h>  // 模块内部使用的库

// ✅ 正确做法
extern int global_var;              // 声明
void helper(void);                  // 声明
static inline int max(int a, int b) { return a > b ? a : b; }  // inline可以
```

### 3.3 前向声明减少依赖

```c
// bad.h — 包含了不必要的头文件
#include "database.h"  // 只为了用 Database* 类型

void process(Database *db);

// good.h — 使用前向声明
struct Database;  // 前向声明，不需要#include

void process(struct Database *db);

// 好处：
// 1. 减少编译依赖（修改database.h不需要重编译包含good.h的文件）
// 2. 加快编译速度
// 3. 避免循环依赖
```

### 3.4 头文件的包含顺序

```c
// 推荐的包含顺序（Google C++ Style Guide的建议也适用于C）
#include "mymodule.h"     // 1. 自己的头文件（确保自包含）
#include <stdio.h>        // 2. 标准C库
#include <stdlib.h>
#include <sys/socket.h>   // 3. 系统头文件
#include "utils.h"        // 4. 项目内其他头文件
#include "third_party.h"  // 5. 第三方库头文件
```

## 四、extern与static关键字

### 4.1 extern的用法

```c
// extern声明：告诉编译器"这个符号在别处定义"

// globals.c — 定义
int shared_counter = 0;
const char *app_name = "MyApp";

// globals.h — 声明
extern int shared_counter;
extern const char *app_name;

// main.c — 使用
#include "globals.h"
void increment() {
    shared_counter++;
}
```

### 4.2 static的四种用法

```c
// 1. static全局变量：限制为文件内部可见
// file1.c
static int internal_state = 0;  // 只在file1.c中可见
// file2.c中无法访问file1.c的internal_state

// 2. static局部变量：生命周期延长到整个程序
void counter() {
    static int count = 0;  // 只初始化一次，每次调用保留上次的值
    count++;
    printf("调用次数: %d\n", count);
}

// 3. static函数：限制为文件内部可见
// helpers.c
static void internal_helper() {  // 只在helpers.c中可见
    // ...
}
void public_func() {
    internal_helper();  // 内部调用OK
}
// 其他文件无法调用internal_helper

// 4. static inline函数（头文件中）
static inline int min(int a, int b) {
    return a < b ? a : b;
}
// 每个包含此头文件的翻译单元都有自己的副本
```

### 4.3 static的信息隐藏作用

```c
// list.c — 实现文件
#include "list.h"

// 内部数据结构（对外隐藏）
static Node *head = NULL;
static int count = 0;

// 内部辅助函数（对外隐藏）
static Node *find_node(int value) {
    Node *curr = head;
    while (curr && curr->data != value) {
        curr = curr->next;
    }
    return curr;
}

// 公开接口
void list_add(int value) {
    Node *node = malloc(sizeof(Node));
    node->data = value;
    node->next = head;
    head = node;
    count++;
}

int list_count() {
    return count;
}
```

## 五、链接属性详解

### 5.1 三种链接属性

```c
// 1. 外部链接（External Linkage）
// 在所有翻译单元中引用同一个实体
int global_var;            // 外部链接
void global_func() {}     // 外部链接
extern int declared_var;   // 声明外部链接符号

// 2. 内部链接（Internal Linkage）
// 只在当前翻译单元中可见
static int file_var;       // 内部链接
static void file_func() {} // 内部链接

// 3. 无链接（No Linkage）
// 局部变量
void func() {
    int local_var;         // 无链接
    static int static_local; // 无链接（但有静态存储期）
}
```

### 5.2 链接属性决定规则

```
标识符                          链接属性      存储期
─────────────────────────────────────────────────────
全局变量                        外部链接      静态
static全局变量                  内部链接      静态
extern声明                      外部链接      (声明)
全局函数                        外部链接      —
static函数                      内部链接      —
局部变量                        无链接        自动
static局部变量                  无链接        静态
函数参数                        无链接        自动
```

### 5.3 ODR（One Definition Rule）

```c
// 一个定义规则：每个符号在所有翻译单元中只能有一个定义

// 错误：在两个.c文件中定义同名全局变量
// a.c: int shared = 1;
// b.c: int shared = 2;
// 链接错误：multiple definition of 'shared'

// 正确做法：
// a.c: int shared = 1;         (定义)
// b.c: extern int shared;      (声明)

// static变量不受ODR约束（因为是内部链接）
// a.c: static int private = 1;  // a.c自己的private
// b.c: static int private = 2;  // b.c自己的private
// 没有冲突，它们是不同的变量
```

## 六、模块化设计模式

### 6.1 模块的基本结构

```c
// ========== stack.h ==========
#ifndef STACK_H
#define STACK_H

#include <stdbool.h>

// 不透明类型
typedef struct Stack Stack;

// 接口
Stack *stack_create(int capacity);
void   stack_destroy(Stack *s);
bool   stack_push(Stack *s, int value);
bool   stack_pop(Stack *s, int *value);
bool   stack_peek(const Stack *s, int *value);
int    stack_size(const Stack *s);
bool   stack_is_empty(const Stack *s);
bool   stack_is_full(const Stack *s);

#endif

// ========== stack.c ==========
#include "stack.h"
#include <stdlib.h>

// 内部实现（对外隐藏）
struct Stack {
    int *data;
    int top;
    int capacity;
};

Stack *stack_create(int capacity) {
    Stack *s = malloc(sizeof(Stack));
    if (!s) return NULL;
    s->data = malloc(capacity * sizeof(int));
    if (!s->data) { free(s); return NULL; }
    s->top = -1;
    s->capacity = capacity;
    return s;
}

void stack_destroy(Stack *s) {
    if (s) {
        free(s->data);
        free(s);
    }
}

bool stack_push(Stack *s, int value) {
    if (stack_is_full(s)) return false;
    s->data[++s->top] = value;
    return true;
}

bool stack_pop(Stack *s, int *value) {
    if (stack_is_empty(s)) return false;
    *value = s->data[s->top--];
    return true;
}

bool stack_peek(const Stack *s, int *value) {
    if (stack_is_empty(s)) return false;
    *value = s->data[s->top];
    return true;
}

int stack_size(const Stack *s) { return s->top + 1; }
bool stack_is_empty(const Stack *s) { return s->top < 0; }
bool stack_is_full(const Stack *s) { return s->top >= s->capacity - 1; }
```

### 6.2 命名空间约定

```c
// C语言没有命名空间，用前缀模拟

// 模块: list
void list_init(List *l);
void list_add(List *l, int value);
int  list_size(const List *l);

// 模块: hashmap
void hashmap_init(HashMap *m);
void hashmap_put(HashMap *m, const char *key, int value);
int  hashmap_get(HashMap *m, const char *key);

// 模块: logger
void logger_init(const char *filename);
void logger_info(const char *fmt, ...);
void logger_error(const char *fmt, ...);

// 好处：
// 1. 避免命名冲突
// 2. 一眼看出函数属于哪个模块
// 3. IDE自动补全更方便
```

## 七、Opaque Pointer模式（信息隐藏）

### 7.1 基本原理

```c
// 头文件中只声明类型，不暴露结构体定义
// 用户只能通过API操作对象，不知道内部细节

// database.h
typedef struct Database Database;  // 不透明类型

Database *db_open(const char *path);
void      db_close(Database *db);
int       db_query(Database *db, const char *sql);

// database.c
struct Database {
    int fd;
    char *path;
    void *cache;
    int transaction_level;
    // 用户完全不知道这些字段
};

Database *db_open(const char *path) {
    Database *db = malloc(sizeof(Database));
    // 初始化内部字段...
    return db;
}
```

### 7.2 Opaque Pointer的优势

```
1. 封装：用户无法直接访问内部成员
   → 防止误操作，API是唯一的访问途径

2. ABI稳定：修改结构体内部不影响使用者
   → 添加新字段不需要重新编译使用者代码
   → 适合动态库

3. 减少编译依赖：用户不需要知道内部依赖
   → database.c可以使用MySQL头文件
   → 但database.h不需要包含MySQL头文件

4. 类似OOP：对象的状态和行为被封装
```

### 7.3 完整的Opaque Pointer示例

```c
// ========== config.h ==========
#ifndef CONFIG_H
#define CONFIG_H

typedef struct Config Config;

Config *config_load(const char *filename);
void    config_save(Config *cfg, const char *filename);
void    config_destroy(Config *cfg);

const char *config_get_string(Config *cfg, const char *key, const char *default_val);
int         config_get_int(Config *cfg, const char *key, int default_val);
void        config_set_string(Config *cfg, const char *key, const char *value);
void        config_set_int(Config *cfg, const char *key, int value);

#endif

// ========== config.c ==========
#include "config.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_ENTRIES 256

typedef struct {
    char key[64];
    char value[256];
} ConfigEntry;

struct Config {
    ConfigEntry entries[MAX_ENTRIES];
    int count;
    char filename[256];
};

Config *config_load(const char *filename) {
    Config *cfg = calloc(1, sizeof(Config));
    strncpy(cfg->filename, filename, sizeof(cfg->filename) - 1);
    
    FILE *fp = fopen(filename, "r");
    if (fp) {
        char line[320];
        while (fgets(line, sizeof(line), fp) && cfg->count < MAX_ENTRIES) {
            char *eq = strchr(line, '=');
            if (eq) {
                *eq = '\0';
                ConfigEntry *e = &cfg->entries[cfg->count++];
                strncpy(e->key, line, sizeof(e->key) - 1);
                // 去除换行
                char *val = eq + 1;
                val[strcspn(val, "\n")] = '\0';
                strncpy(e->value, val, sizeof(e->value) - 1);
            }
        }
        fclose(fp);
    }
    return cfg;
}

static ConfigEntry *find_entry(Config *cfg, const char *key) {
    for (int i = 0; i < cfg->count; i++) {
        if (strcmp(cfg->entries[i].key, key) == 0) {
            return &cfg->entries[i];
        }
    }
    return NULL;
}

const char *config_get_string(Config *cfg, const char *key, const char *def) {
    ConfigEntry *e = find_entry(cfg, key);
    return e ? e->value : def;
}

int config_get_int(Config *cfg, const char *key, int def) {
    ConfigEntry *e = find_entry(cfg, key);
    return e ? atoi(e->value) : def;
}

void config_set_string(Config *cfg, const char *key, const char *value) {
    ConfigEntry *e = find_entry(cfg, key);
    if (e) {
        strncpy(e->value, value, sizeof(e->value) - 1);
    } else if (cfg->count < MAX_ENTRIES) {
        e = &cfg->entries[cfg->count++];
        strncpy(e->key, key, sizeof(e->key) - 1);
        strncpy(e->value, value, sizeof(e->value) - 1);
    }
}

void config_set_int(Config *cfg, const char *key, int value) {
    char buf[32];
    snprintf(buf, sizeof(buf), "%d", value);
    config_set_string(cfg, key, buf);
}

void config_save(Config *cfg, const char *filename) {
    FILE *fp = fopen(filename, "w");
    if (fp) {
        for (int i = 0; i < cfg->count; i++) {
            fprintf(fp, "%s=%s\n", cfg->entries[i].key, cfg->entries[i].value);
        }
        fclose(fp);
    }
}

void config_destroy(Config *cfg) {
    free(cfg);
}
```

## 八、接口与实现分离

### 8.1 抽象接口（类似接口/虚函数表）

```c
// ========== storage.h ==========
typedef struct Storage Storage;

// "虚函数表"
typedef struct {
    int  (*open)(Storage *s, const char *path);
    void (*close)(Storage *s);
    int  (*read)(Storage *s, const char *key, char *buf, size_t size);
    int  (*write)(Storage *s, const char *key, const char *data, size_t size);
    void (*destroy)(Storage *s);
} StorageOps;

struct Storage {
    const StorageOps *ops;  // 指向操作函数表
    void *impl;             // 指向具体实现的数据
};

// 通用接口
static inline int storage_read(Storage *s, const char *key, char *buf, size_t size) {
    return s->ops->read(s, key, buf, size);
}

static inline int storage_write(Storage *s, const char *key, const char *data, size_t size) {
    return s->ops->write(s, key, data, size);
}

// 工厂函数
Storage *file_storage_create(void);
Storage *memory_storage_create(void);
```

```c
// ========== file_storage.c ==========
#include "storage.h"

typedef struct {
    char base_path[256];
} FileStorageData;

static int file_open(Storage *s, const char *path) {
    FileStorageData *data = s->impl;
    strncpy(data->base_path, path, sizeof(data->base_path) - 1);
    return 0;
}

static int file_read(Storage *s, const char *key, char *buf, size_t size) {
    FileStorageData *data = s->impl;
    char filepath[512];
    snprintf(filepath, sizeof(filepath), "%s/%s", data->base_path, key);
    FILE *fp = fopen(filepath, "r");
    if (!fp) return -1;
    size_t n = fread(buf, 1, size - 1, fp);
    buf[n] = '\0';
    fclose(fp);
    return (int)n;
}

// ... 其他实现 ...

static const StorageOps file_ops = {
    .open = file_open,
    .close = file_close,
    .read = file_read,
    .write = file_write,
    .destroy = file_destroy
};

Storage *file_storage_create(void) {
    Storage *s = malloc(sizeof(Storage));
    s->ops = &file_ops;
    s->impl = calloc(1, sizeof(FileStorageData));
    return s;
}
```

## 九、单例模式与全局状态管理

### 9.1 简单的单例

```c
// logger.c
#include "logger.h"
#include <stdio.h>

static FILE *log_file = NULL;
static int log_level = LOG_INFO;

void logger_init(const char *filename, int level) {
    if (log_file) return;  // 已经初始化
    log_file = fopen(filename, "a");
    log_level = level;
}

void logger_write(int level, const char *fmt, ...) {
    if (!log_file || level < log_level) return;
    
    va_list args;
    va_start(args, fmt);
    vfprintf(log_file, fmt, args);
    va_end(args);
    fprintf(log_file, "\n");
    fflush(log_file);
}

void logger_cleanup(void) {
    if (log_file) {
        fclose(log_file);
        log_file = NULL;
    }
}
```

### 9.2 线程安全的初始化

```c
#include <pthread.h>

static pthread_once_t init_once = PTHREAD_ONCE_INIT;
static Logger *instance = NULL;

static void do_init(void) {
    instance = malloc(sizeof(Logger));
    // 初始化...
}

Logger *logger_get_instance(void) {
    pthread_once(&init_once, do_init);
    return instance;
}
```

## 十、回调机制与插件架构

### 10.1 回调函数

```c
// 事件系统
typedef void (*EventHandler)(const char *event, void *data, void *userdata);

typedef struct {
    char event_name[32];
    EventHandler handler;
    void *userdata;
} EventBinding;

#define MAX_BINDINGS 100
static EventBinding bindings[MAX_BINDINGS];
static int binding_count = 0;

void event_on(const char *event, EventHandler handler, void *userdata) {
    if (binding_count < MAX_BINDINGS) {
        EventBinding *b = &bindings[binding_count++];
        strncpy(b->event_name, event, sizeof(b->event_name) - 1);
        b->handler = handler;
        b->userdata = userdata;
    }
}

void event_emit(const char *event, void *data) {
    for (int i = 0; i < binding_count; i++) {
        if (strcmp(bindings[i].event_name, event) == 0) {
            bindings[i].handler(event, data, bindings[i].userdata);
        }
    }
}

// 使用
void on_user_login(const char *event, void *data, void *userdata) {
    const char *username = (const char*)data;
    printf("用户登录: %s\n", username);
}

event_on("login", on_user_login, NULL);
event_emit("login", "alice");
```

#include <dlfcn.h>
int plugin_load(const char *path) {
    void *handle = dlopen(path, RTLD_LAZY);
    if (!handle) return -1;
    
    typedef const Plugin *(*GetPluginFunc)(void);
    GetPluginFunc get_plugin = dlsym(handle, "get_plugin");
    if (!get_plugin) { dlclose(handle); return -1; }
    
    const Plugin *p = get_plugin();
    plugin_register(p);
    p->init();
    return 0;
}

### 11.3 goto cleanup模式

```c
int process_file(const char *input, const char *output) {
    int ret = -1;
    FILE *fin = NULL;
    FILE *fout = NULL;
    char *buffer = NULL;
    
    fin = fopen(input, "r");
    if (!fin) goto cleanup;
    
    fout = fopen(output, "w");
    if (!fout) goto cleanup;
    
    buffer = malloc(4096);
    if (!buffer) goto cleanup;
    
    // 业务逻辑
    size_t n;
    while ((n = fread(buffer, 1, 4096, fin)) > 0) {
        fwrite(buffer, 1, n, fout);
    }
    
    ret = 0;  // 成功

cleanup:
    free(buffer);
    if (fout) fclose(fout);
    if (fin) fclose(fin);
    return ret;
}
```

## 十二、Makefile与构建管理

### 12.1 项目级Makefile

```makefile
# 项目配置
CC = gcc
CFLAGS = -Wall -Wextra -Werror -std=c11 -g
LDFLAGS = -lm
INCLUDES = -Iinclude

# 目录
SRC_DIR = src
OBJ_DIR = build
BIN_DIR = bin

# 源文件和目标文件
SRCS = $(wildcard $(SRC_DIR)/*.c)
OBJS = $(patsubst $(SRC_DIR)/%.c, $(OBJ_DIR)/%.o, $(SRCS))
TARGET = $(BIN_DIR)/myapp

# 默认目标
all: dirs $(TARGET)

# 创建目录
dirs:
	@mkdir -p $(OBJ_DIR) $(BIN_DIR)

# 链接
$(TARGET): $(OBJS)
	$(CC) $(OBJS) -o $@ $(LDFLAGS)

# 编译
$(OBJ_DIR)/%.o: $(SRC_DIR)/%.c
	$(CC) $(CFLAGS) $(INCLUDES) -c $< -o $@

# 自动依赖生成
-include $(OBJS:.o=.d)
$(OBJ_DIR)/%.d: $(SRC_DIR)/%.c
	@$(CC) $(CFLAGS) $(INCLUDES) -MM $< -MT $(OBJ_DIR)/$*.o > $@

# 清理
clean:
	rm -rf $(OBJ_DIR) $(BIN_DIR)

# 调试版本
debug: CFLAGS += -DDEBUG -O0
debug: all

# 发布版本
release: CFLAGS += -DNDEBUG -O2
release: all

.PHONY: all clean dirs debug release
```

## 十三、常见问题与最佳实践

### 13.1 循环依赖的解决

```c
// a.h 需要 B 的类型，b.h 需要 A 的类型 → 循环依赖

// 解决方案：使用前向声明
// a.h
#ifndef A_H
#define A_H
struct B;  // 前向声明
typedef struct {
    int value;
    struct B *partner;  // 只用指针，不需要完整定义
} A;
void a_func(A *a, struct B *b);
#endif

// b.h
#ifndef B_H
#define B_H
struct A;  // 前向声明
typedef struct B {
    int value;
    struct A *partner;
} B;
void b_func(B *b, struct A *a);
#endif
```

### 13.2 最佳实践清单

```
1. 每个.c文件有对应的.h文件
2. 头文件使用include guard或#pragma once
3. 内部函数和变量用static修饰
4. 头文件中只放声明，不放定义
5. 使用一致的命名前缀（模块名_函数名）
6. Opaque Pointer隐藏实现细节
7. 每个模块有init和cleanup函数
8. 使用goto cleanup处理错误和资源释放
9. 编译时开启所有警告（-Wall -Wextra）
10. 保持头文件最小化，使用前向声明减少依赖
```

## 一、引言：C语言如何与外部世界交互

### 1.1 "一切皆文件"

在Unix/Linux哲学中，几乎所有IO资源都被抽象为文件：

```
普通文件      /home/user/data.txt
目录文件      /home/user/
设备文件      /dev/sda（块设备）, /dev/tty（字符设备）
管道文件      pipe创建的匿名管道, 命名管道(FIFO)
套接字文件    /tmp/mysql.sock
符号链接      /usr/bin/python → python3
/proc文件    /proc/cpuinfo（虚拟文件系统）
```

### 1.2 两层IO接口

```
┌─────────────────────────────────────┐
│           应用程序                    │
├─────────────────────────────────────┤
│   标准IO库（stdio.h）               │  ← 带缓冲的高级接口
│   fopen, fread, fwrite, fprintf...  │
├─────────────────────────────────────┤
│   系统调用接口                       │  ← 无缓冲的低级接口
│   open, read, write, close...       │
├─────────────────────────────────────┤
│   Linux内核                          │
│   VFS → 文件系统 → 块设备驱动        │
└─────────────────────────────────────┘
```

## 二、文件描述符与内核文件系统

### 2.1 文件描述符（fd）

```c
// 文件描述符是一个小的非负整数
// 每个进程都有一个文件描述符表

// 预定义的文件描述符：
// 0 = stdin  (标准输入)
// 1 = stdout (标准输出)
// 2 = stderr (标准错误)

int fd = open("test.txt", O_RDONLY);
// fd通常从3开始（0,1,2已被占用）

printf("打开的文件描述符: %d\n", fd);
close(fd);
```

### 2.2 内核中的文件管理结构

```
进程级:
  struct task_struct {
      struct files_struct *files;  → 文件描述符表
  }

  文件描述符表:
  ┌────┬──────────────┐
  │ fd │ struct file * │
  ├────┼──────────────┤
  │  0 │ → stdin文件对象 │
  │  1 │ → stdout文件对象│
  │  2 │ → stderr文件对象│
  │  3 │ → test.txt文件对象│
  └────┴──────────────┘

系统级:
  struct file {
      loff_t f_pos;           // 文件偏移量
      const struct file_operations *f_op;  // 操作函数表
      struct inode *f_inode;  // inode指针
      unsigned int f_flags;   // 打开标志
      atomic_t f_count;       // 引用计数
  }

文件系统级:
  struct inode {
      umode_t i_mode;         // 权限和类型
      uid_t i_uid;            // 所有者
      loff_t i_size;          // 文件大小
      struct timespec i_mtime; // 修改时间
      unsigned long i_ino;    // inode编号
  }
```

### 2.3 文件描述符的继承与共享

```c
#include <unistd.h>
#include <fcntl.h>

int main() {
    int fd = open("shared.txt", O_RDWR | O_CREAT, 0644);
    write(fd, "Hello", 5);
    
    pid_t pid = fork();
    if (pid == 0) {
        // 子进程继承了fd，共享同一个struct file
        // 偏移量也是共享的！
        write(fd, " Child", 6);
        close(fd);
        _exit(0);
    } else {
        wait(NULL);
        write(fd, " Parent", 7);
        close(fd);
    }
    
    // shared.txt内容: "Hello Child Parent"
    // 因为父子进程共享文件偏移量
    return 0;
}
```

## 三、系统调用：用户态与内核态

### 3.1 系统调用的过程

```
用户态代码: write(fd, buf, count)
    │
    ↓ libc包装函数
设置系统调用号到寄存器(rax = __NR_write = 1)
设置参数(rdi=fd, rsi=buf, rdx=count)
执行 syscall 指令
    │
    ↓ CPU特权级切换: Ring 3 → Ring 0
    │
内核态: sys_write(fd, buf, count)
    │
    ├─ 查找fd对应的struct file
    ├─ 调用文件系统的write方法
    ├─ 数据写入页缓存（Page Cache）
    └─ 返回写入字节数
    │
    ↓ CPU特权级切换: Ring 0 → Ring 3
    │
用户态: 接收返回值
```

### 3.2 系统调用的开销

```c
// 系统调用的开销约为几百个CPU时钟周期
// 主要开销来自：
// 1. 用户态/内核态切换
// 2. 保存和恢复寄存器上下文
// 3. TLB刷新（某些情况）
// 4. 安全检查

// 因此，标准IO库引入了缓冲机制
// 将多次小写入合并为一次大的系统调用
// 大幅减少系统调用次数

// 实验：比较系统调用次数
// 方式1：每次写1字节（系统调用密集）
for (int i = 0; i < 1000000; i++) {
    write(fd, "a", 1);  // 100万次系统调用
}

// 方式2：一次写入（1次系统调用）
char buf[1000000];
memset(buf, 'a', sizeof(buf));
write(fd, buf, sizeof(buf));  // 1次系统调用

// 速度差异可达100倍以上
```

### 3.3 用strace追踪系统调用

```bash
# 追踪程序的所有系统调用
strace ./program

# 只追踪文件相关的系统调用
strace -e trace=file ./program

# 统计系统调用次数
strace -c ./program

# 输出示例：
# % time     seconds  usecs/call     calls    syscall
# ------ ----------- ----------- --------- ---------
#  80.00    0.002400           2      1000  write
#  10.00    0.000300          30        10  open
#   5.00    0.000150          15        10  close
#   5.00    0.000150           0      1000  read
```

## 四、底层IO系统调用

### 4.1 open/close

```c
#include <fcntl.h>
#include <unistd.h>

// 打开文件
int fd = open("test.txt", O_RDONLY);
int fd2 = open("new.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);
int fd3 = open("log.txt", O_WRONLY | O_CREAT | O_APPEND, 0644);

// 打开标志：
// O_RDONLY   只读
// O_WRONLY   只写
// O_RDWR     读写
// O_CREAT    不存在则创建
// O_TRUNC    存在则截断为0
// O_APPEND   追加模式
// O_NONBLOCK 非阻塞模式
// O_EXCL     与O_CREAT一起使用，文件存在则失败（原子创建）

// 权限位（mode）：当O_CREAT时需要
// 0644 = rw-r--r--
// 实际权限 = mode & ~umask

// 关闭文件
close(fd);
```

### 4.2 read/write

```c
#include <unistd.h>

char buf[4096];

// 读取
ssize_t n = read(fd, buf, sizeof(buf));
// 返回值:
// > 0: 实际读取的字节数
// = 0: 到达文件末尾(EOF)
// < 0: 错误（errno被设置）

// 写入
ssize_t written = write(fd, "Hello, World!\n", 14);
// 返回值:
// > 0: 实际写入的字节数（可能小于请求的）
// < 0: 错误

// 可靠的读写函数（处理部分读/写）
ssize_t readn(int fd, void *buf, size_t count) {
    size_t nleft = count;
    ssize_t nread;
    char *ptr = buf;
    
    while (nleft > 0) {
        nread = read(fd, ptr, nleft);
        if (nread < 0) {
            if (errno == EINTR) continue;  // 被信号中断，重试
            return -1;
        }
        if (nread == 0) break;  // EOF
        nleft -= nread;
        ptr += nread;
    }
    return count - nleft;
}

ssize_t writen(int fd, const void *buf, size_t count) {
    size_t nleft = count;
    ssize_t nwritten;
    const char *ptr = buf;
    
    while (nleft > 0) {
        nwritten = write(fd, ptr, nleft);
        if (nwritten < 0) {
            if (errno == EINTR) continue;
            return -1;
        }
        nleft -= nwritten;
        ptr += nwritten;
    }
    return count;
}
```

### 4.3 lseek

```c
#include <unistd.h>

// 移动文件偏移量
off_t pos = lseek(fd, 0, SEEK_SET);    // 移到文件开头
off_t pos2 = lseek(fd, 100, SEEK_SET); // 移到偏移100
off_t pos3 = lseek(fd, -10, SEEK_END); // 移到距文件末尾10字节
off_t cur = lseek(fd, 0, SEEK_CUR);    // 获取当前偏移量

// SEEK_SET: 相对于文件开头
// SEEK_CUR: 相对于当前位置
// SEEK_END: 相对于文件末尾

// 创建空洞文件
int fd = open("sparse.dat", O_WRONLY | O_CREAT | O_TRUNC, 0644);
write(fd, "Hello", 5);
lseek(fd, 1000000, SEEK_SET);  // 跳到1MB位置
write(fd, "World", 5);
close(fd);
// 文件逻辑大小约1MB，但实际只占用了2个磁盘块
```

### 4.4 dup/dup2（文件描述符复制）

```c
#include <unistd.h>
#include <fcntl.h>

// 重定向标准输出到文件
int fd = open("output.txt", O_WRONLY | O_CREAT | O_TRUNC, 0644);

int old_stdout = dup(STDOUT_FILENO);  // 保存原来的stdout
dup2(fd, STDOUT_FILENO);              // 将fd复制到1（stdout）
close(fd);                             // 关闭多余的fd

printf("这行会写入output.txt\n");

// 恢复stdout
dup2(old_stdout, STDOUT_FILENO);
close(old_stdout);

printf("这行会显示在终端\n");
```

## 五、标准IO库（stdio）

### 5.1 FILE结构

```c
// FILE是stdio的核心抽象，封装了文件描述符和缓冲区
// 简化的FILE结构（glibc实际实现更复杂）
struct _IO_FILE {
    int _fileno;              // 底层文件描述符
    char *_IO_buf_base;       // 缓冲区起始
    char *_IO_buf_end;        // 缓冲区结束
    char *_IO_read_ptr;       // 当前读位置
    char *_IO_write_ptr;      // 当前写位置
    int _flags;               // 状态标志
    // ...
};
typedef struct _IO_FILE FILE;

// 预定义的FILE对象
// stdin  → 文件描述符0
// stdout → 文件描述符1
// stderr → 文件描述符2
```

### 5.2 fopen/fclose

```c
#include <stdio.h>

// 打开文件
FILE *fp = fopen("test.txt", "r");    // 只读
FILE *fp2 = fopen("new.txt", "w");    // 写入（截断）
FILE *fp3 = fopen("log.txt", "a");    // 追加
FILE *fp4 = fopen("data.bin", "rb");  // 二进制只读
FILE *fp5 = fopen("data.bin", "w+b"); // 二进制读写（截断）

// 打开模式：
// "r"   只读，文件必须存在
// "w"   写入，创建或截断
// "a"   追加，创建或追加
// "r+"  读写，文件必须存在
// "w+"  读写，创建或截断
// "a+"  读追加，创建或追加
// 加"b" 二进制模式（Windows有区别，Linux无区别）

if (fp == NULL) {
    perror("fopen");  // 打印错误信息
    return 1;
}

// 关闭文件
fclose(fp);
// fclose会:
// 1. 冲刷缓冲区（写入剩余数据）
// 2. 释放缓冲区内存
// 3. 关闭底层文件描述符
```

### 5.3 字符/行/块IO

```c
// 字符IO
int ch = fgetc(fp);       // 读一个字符
fputc('A', fp);            // 写一个字符

// 行IO
char line[256];
fgets(line, sizeof(line), fp);  // 读一行（包含\n）
fputs("Hello\n", fp);          // 写一行

// 块IO
struct Data records[100];
size_t n = fread(records, sizeof(struct Data), 100, fp);
fwrite(records, sizeof(struct Data), n, fp);

// fread返回成功读取的元素个数（不是字节数）
// 如果返回值 < 请求数量：
//   feof(fp) 为真 → 到达文件末尾
//   ferror(fp) 为真 → 发生错误
```

## 六、缓冲机制详解

### 6.1 三种缓冲模式

```c
// 1. 全缓冲（Full Buffering）
// 缓冲区满了才冲刷（写入底层fd）
// 默认用于普通文件
FILE *fp = fopen("file.txt", "w");  // 全缓冲

// 2. 行缓冲（Line Buffering）
// 遇到换行符时冲刷
// 默认用于连接终端的流
stdout;  // 连接终端时是行缓冲

// 3. 无缓冲（Unbuffered）
// 每次IO操作都直接写
stderr;  // 标准错误流是无缓冲

// 修改缓冲模式
setvbuf(fp, NULL, _IOFBF, 8192);  // 全缓冲，8KB
setvbuf(fp, NULL, _IOLBF, 1024);  // 行缓冲，1KB
setvbuf(fp, NULL, _IONBF, 0);     // 无缓冲

// 手动冲刷
fflush(fp);     // 冲刷指定流
fflush(NULL);   // 冲刷所有打开的输出流
```

### 6.2 缓冲的工作原理

```
写入过程（全缓冲）:

应用程序调用 fwrite(data, 1, 100, fp)
    │
    ↓
将100字节复制到FILE内部缓冲区
    │
    ├── 缓冲区未满 → 返回（不调用write系统调用）
    │
    └── 缓冲区满了 → 调用write(fd, buffer, bufsize)
                     清空缓冲区
                     将剩余数据放入缓冲区

冲刷触发条件：
1. 缓冲区满
2. fflush显式调用
3. fclose关闭文件
4. 进程正常退出（调用exit）
5. 行缓冲遇到\n
6. 从无缓冲/行缓冲的流读取输入前
```

### 6.3 缓冲带来的性能提升

```c
#include <stdio.h>
#include <time.h>
#include <unistd.h>

#define COUNT 1000000

// 方式1：每次write一个字节（无缓冲）
void test_unbuffered() {
    int fd = open("test1.dat", O_WRONLY | O_CREAT | O_TRUNC, 0644);
    for (int i = 0; i < COUNT; i++) {
        write(fd, "a", 1);  // 100万次系统调用
    }
    close(fd);
}

// 方式2：使用stdio缓冲
void test_buffered() {
    FILE *fp = fopen("test2.dat", "w");
    for (int i = 0; i < COUNT; i++) {
        fputc('a', fp);  // 大部分只是内存复制
    }
    fclose(fp);
}

// 方式2可能比方式1快50-100倍
```

### 6.4 缓冲的陷阱

```c
// 陷阱1：fork后子进程继承缓冲区
printf("Hello ");  // 还在缓冲区中
fork();
// 父子进程各有一份缓冲区，"Hello "可能被输出两次

// 解决：fork前冲刷
fflush(stdout);
fork();

// 陷阱2：混用stdio和系统调用
fprintf(fp, "stdio写入");  // 数据在缓冲区
write(fileno(fp), "系统调用写入", 18);  // 直接写入fd
// 输出顺序可能不符合预期

// 陷阱3：程序异常退出时缓冲区数据丢失
printf("重要数据");  // 还在缓冲区
abort();  // 缓冲区数据丢失！
// 解决：使用write直接写，或者fflush后再做危险操作
```

## 七、格式化IO的实现原理

### 7.1 printf家族

```c
// 输出到stdout
printf("值: %d\n", 42);

// 输出到文件
fprintf(fp, "值: %d\n", 42);

// 输出到字符串
char buf[100];
snprintf(buf, sizeof(buf), "值: %d", 42);

// 输出到动态分配的字符串（GNU扩展）
char *str;
asprintf(&str, "值: %d", 42);
free(str);
```

### 7.2 printf格式说明符

```c
// 基本格式: %[flags][width][.precision][length]specifier

// 说明符
printf("%d\n", 42);          // 十进制整数: 42
printf("%u\n", 42u);         // 无符号十进制: 42
printf("%x\n", 255);         // 十六进制: ff
printf("%o\n", 8);           // 八进制: 10
printf("%f\n", 3.14);        // 浮点: 3.140000
printf("%e\n", 3.14);        // 科学计数: 3.140000e+00
printf("%s\n", "hello");     // 字符串: hello
printf("%c\n", 'A');         // 字符: A
printf("%p\n", &x);          // 指针: 0x7ffd1234
printf("%%\n");              // 百分号: %

// 标志
printf("%-10d|\n", 42);      // 左对齐: "42        |"
printf("%+d\n", 42);         // 显示正号: +42
printf("%010d\n", 42);       // 零填充: 0000000042
printf("%#x\n", 255);        // 加前缀: 0xff

// 宽度和精度
printf("%10d\n", 42);        // 宽度10: "        42"
printf("%.5f\n", 3.14);      // 精度5: "3.14000"
printf("%.3s\n", "Hello");   // 截断: "Hel"
printf("%*d\n", 10, 42);     // 动态宽度: "        42"

// 长度修饰符
printf("%ld\n", 42L);        // long
printf("%lld\n", 42LL);      // long long
printf("%zu\n", sizeof(int)); // size_t
printf("%td\n", p2 - p1);    // ptrdiff_t
```

### 7.3 scanf家族

```c
int n;
float f;
char str[100];

// 从stdin读取
scanf("%d %f %99s", &n, &f, str);

// 从文件读取
fscanf(fp, "%d %f %99s", &n, &f, str);

// 从字符串读取
sscanf("42 3.14 hello", "%d %f %99s", &n, &f, str);

// 安全注意：
// 1. %s不限制长度 → 缓冲区溢出
scanf("%s", str);       // 危险！
scanf("%99s", str);     // 安全：限制99字符

// 2. 检查返回值
int ret = scanf("%d", &n);
if (ret != 1) {
    printf("输入错误\n");
}
```

## 八、文件定位与随机访问

### 8.1 fseek/ftell/rewind

```c
FILE *fp = fopen("data.bin", "rb");

// 获取当前位置
long pos = ftell(fp);  // 返回当前偏移量

// 移动到指定位置
fseek(fp, 0, SEEK_SET);    // 移到开头
fseek(fp, 100, SEEK_SET);  // 移到偏移100
fseek(fp, -10, SEEK_END);  // 从末尾往前10字节
fseek(fp, 50, SEEK_CUR);   // 从当前位置前进50字节

// 回到开头
rewind(fp);  // 等价于 fseek(fp, 0, SEEK_SET); clearerr(fp);
```

### 8.2 获取文件大小

```c
// 方法1：使用fseek/ftell
long get_file_size(const char *filename) {
    FILE *fp = fopen(filename, "rb");
    if (!fp) return -1;
    fseek(fp, 0, SEEK_END);
    long size = ftell(fp);
    fclose(fp);
    return size;
}

// 方法2：使用stat
#include <sys/stat.h>
long get_file_size2(const char *filename) {
    struct stat st;
    if (stat(filename, &st) == 0) {
        return st.st_size;
    }
    return -1;
}
```

### 8.3 二进制文件的随机读写

```c
// 记录型文件的随机访问
typedef struct {
    int id;
    char name[32];
    double score;
} Record;

// 写入记录
void write_record(FILE *fp, int index, const Record *rec) {
    fseek(fp, index * sizeof(Record), SEEK_SET);
    fwrite(rec, sizeof(Record), 1, fp);
}

// 读取记录
void read_record(FILE *fp, int index, Record *rec) {
    fseek(fp, index * sizeof(Record), SEEK_SET);
    fread(rec, sizeof(Record), 1, fp);
}

// 使用
FILE *fp = fopen("records.dat", "w+b");
Record r = {1, "张三", 95.5};
write_record(fp, 0, &r);

Record r2 = {2, "李四", 88.0};
write_record(fp, 1, &r2);

// 随机读取第0条记录
Record result;
read_record(fp, 0, &result);
printf("ID=%d, Name=%s, Score=%.1f\n", result.id, result.name, result.score);
fclose(fp);
```

## 九、目录操作

### 9.1 目录读取

```c
#include <dirent.h>
#include <stdio.h>

void list_directory(const char *path) {
    DIR *dir = opendir(path);
    if (dir == NULL) {
        perror("opendir");
        return;
    }
    
    struct dirent *entry;
    while ((entry = readdir(dir)) != NULL) {
        // 跳过 . 和 ..
        if (strcmp(entry->d_name, ".") == 0 || 
            strcmp(entry->d_name, "..") == 0) {
            continue;
        }
        
        printf("%s", entry->d_name);
        
        switch (entry->d_type) {
            case DT_REG: printf(" [文件]"); break;
            case DT_DIR: printf(" [目录]"); break;
            case DT_LNK: printf(" [链接]"); break;
            default:     printf(" [其他]"); break;
        }
        printf("\n");
    }
    
    closedir(dir);
}
```

### 9.2 递归遍历目录

```c
#include <dirent.h>
#include <sys/stat.h>
#include <string.h>

void walk_directory(const char *path, int depth) {
    DIR *dir = opendir(path);
    if (!dir) return;
    
    struct dirent *entry;
    while ((entry = readdir(dir)) != NULL) {
        if (entry->d_name[0] == '.') continue;
        
        char fullpath[PATH_MAX];
        snprintf(fullpath, sizeof(fullpath), "%s/%s", path, entry->d_name);
        
        struct stat st;
        if (stat(fullpath, &st) != 0) continue;
        
        // 缩进
        for (int i = 0; i < depth; i++) printf("  ");
        
        if (S_ISDIR(st.st_mode)) {
            printf("[DIR]  %s\n", entry->d_name);
            walk_directory(fullpath, depth + 1);  // 递归
        } else if (S_ISREG(st.st_mode)) {
            printf("[FILE] %s (%ld bytes)\n", entry->d_name, st.st_size);
        }
    }
    
    closedir(dir);
}
```

## 十一、文件锁机制

### 11.1 fcntl文件锁

```c
#include <fcntl.h>

int fd = open("shared.dat", O_RDWR);

// 设置写锁（排他锁）
struct flock lock;
lock.l_type = F_WRLCK;     // F_RDLCK(读锁), F_WRLCK(写锁), F_UNLCK(解锁)
lock.l_whence = SEEK_SET;
lock.l_start = 0;           // 锁定起始偏移
lock.l_len = 0;             // 0表示锁定到文件末尾

// 阻塞式加锁
if (fcntl(fd, F_SETLKW, &lock) == -1) {
    perror("fcntl");
}

// 非阻塞式加锁
if (fcntl(fd, F_SETLK, &lock) == -1) {
    if (errno == EAGAIN || errno == EACCES) {
        printf("文件已被锁定\n");
    }
}

// ... 操作文件 ...

// 解锁
lock.l_type = F_UNLCK;
fcntl(fd, F_SETLK, &lock);
close(fd);
```

## 十二、内存映射IO（mmap）

### 12.1 文件映射

```c
#include <sys/mman.h>
#include <sys/stat.h>
#include <fcntl.h>

int fd = open("data.txt", O_RDWR);
struct stat st;
fstat(fd, &st);

// 将文件映射到内存
char *map = mmap(NULL, st.st_size,
                 PROT_READ | PROT_WRITE,  // 可读可写
                 MAP_SHARED,               // 修改反映到文件
                 fd, 0);                   // 从文件偏移0开始

if (map == MAP_FAILED) {
    perror("mmap");
    close(fd);
    return 1;
}

close(fd);  // 映射后可以关闭fd

// 直接操作内存即等于操作文件
printf("文件内容: %.*s\n", (int)st.st_size, map);

// 修改（MAP_SHARED时会写回文件）
map[0] = 'H';

// 同步到磁盘
msync(map, st.st_size, MS_SYNC);

// 取消映射
munmap(map, st.st_size);
```

### 12.2 mmap vs read/write

```
                read/write                 mmap
数据路径:    磁盘→页缓存→用户缓冲区    磁盘→页缓存（直接映射）
数据拷贝:    需要从内核空间到用户空间    零拷贝（共享物理页）
系统调用:    每次读写都需要               映射后不需要（除了同步）
随机访问:    需要lseek                    直接用指针偏移
适用场景:    顺序读写、小文件             随机访问、大文件、共享内存
线程安全:    每次调用独立                  需要自行同步
```

### 12.3 进程间共享内存

```c
// 父子进程通过mmap共享内存
int *shared = mmap(NULL, sizeof(int),
                   PROT_READ | PROT_WRITE,
                   MAP_SHARED | MAP_ANONYMOUS,
                   -1, 0);

*shared = 0;

pid_t pid = fork();
if (pid == 0) {
    // 子进程
    (*shared)++;
    printf("子进程: shared = %d\n", *shared);
    _exit(0);
} else {
    wait(NULL);
    printf("父进程: shared = %d\n", *shared);  // 1
}

munmap(shared, sizeof(int));
```

## 十三、IO多路复用基础

### 13.1 select

```c
#include <sys/select.h>

fd_set readfds;
FD_ZERO(&readfds);
FD_SET(STDIN_FILENO, &readfds);
FD_SET(sock_fd, &readfds);

struct timeval timeout = {5, 0};  // 5秒超时

int ready = select(sock_fd + 1, &readfds, NULL, NULL, &timeout);
if (ready > 0) {
    if (FD_ISSET(STDIN_FILENO, &readfds)) {
        // 标准输入有数据
    }
    if (FD_ISSET(sock_fd, &readfds)) {
        // socket有数据
    }
}
```

### 13.2 poll

```c
#include <poll.h>

struct pollfd fds[2];
fds[0].fd = STDIN_FILENO;
fds[0].events = POLLIN;
fds[1].fd = sock_fd;
fds[1].events = POLLIN;

int ready = poll(fds, 2, 5000);  // 5秒超时
if (ready > 0) {
    if (fds[0].revents & POLLIN) {
        // stdin可读
    }
    if (fds[1].revents & POLLIN) {
        // socket可读
    }
}
```

### 13.3 epoll（Linux特有）

```c
#include <sys/epoll.h>

int epfd = epoll_create1(0);

struct epoll_event ev;
ev.events = EPOLLIN;
ev.data.fd = sock_fd;
epoll_ctl(epfd, EPOLL_CTL_ADD, sock_fd, &ev);

struct epoll_event events[10];
int n = epoll_wait(epfd, events, 10, 5000);

for (int i = 0; i < n; i++) {
    if (events[i].data.fd == sock_fd) {
        // socket可读
    }
}

close(epfd);
```

### 13.4 三者对比

```
         select              poll                epoll
最大fd:  FD_SETSIZE(1024)   无限制              无限制
实现:    遍历所有fd          遍历所有fd          回调通知
复杂度:  O(n)               O(n)                O(1)
拷贝:    每次调用传整个集合  每次调用传整个数组  只需添加一次
适用:    跨平台、少量fd     跨平台、中量fd      Linux、大量fd
```

## 十四、总结与面试高频问题

### 14.1 核心知识总结

```
文件IO体系：

系统调用层:
  open/close/read/write/lseek/dup/mmap
  → 无缓冲，直接与内核交互

标准IO层:
  fopen/fclose/fread/fwrite/fprintf/fgets
  → 带缓冲，减少系统调用次数

缓冲策略:
  全缓冲（文件）→ 行缓冲（终端）→ 无缓冲（stderr）

高级IO:
  mmap: 内存映射，零拷贝
  fcntl锁: 文件级同步
  IO多路复用: select/poll/epoll
```

### 14.2 面试高频问题

**Q1: stdio的FILE和系统调用的fd什么关系？**
```
FILE是对fd的封装，内部包含fd和用户态缓冲区。
fileno(fp)获取FILE对应的fd。
fdopen(fd, mode)将fd包装为FILE。
```

**Q2: fwrite返回后数据一定到磁盘了吗？**
```
不一定。fwrite只保证数据到了用户态缓冲区。
fflush保证数据到了内核页缓存。
fsync(fileno(fp))才保证数据到了磁盘。
```

**Q3: printf和write可以混用吗？**
```
技术上可以，但可能导致输出顺序混乱。
printf的数据在缓冲区中，write直接写入fd。
如果混用，需要在write前调用fflush(stdout)。
```

**Q4: mmap的优势？**
```
1. 零拷贝（不需要从内核缓冲区到用户缓冲区）
2. 随机访问高效（直接指针操作）
3. 进程间共享（MAP_SHARED）
4. 延迟加载（按需分页）
```

---

> **核心思想**：C语言的IO系统分为两层——底层系统调用和上层标准IO库。系统调用提供了最基本的文件操作能力，而标准IO库通过缓冲机制大幅提升了性能。理解两层的关系、缓冲的工作方式、mmap的原理，是写出高性能IO代码的基础。

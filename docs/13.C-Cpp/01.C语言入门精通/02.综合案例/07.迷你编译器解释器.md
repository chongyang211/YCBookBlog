# 第七章：C语言 迷你编译器解释器

本章是综合案例的**第七关·毕业设计**——用 C 语言实现一个四则运算表达式解释器，走完 **Lexer（词法）→ Parser（语法）→ AST（抽象语法树）→ Interpreter（解释执行）** 的编译原理五段式前四段。不依赖任何第三方库，纯 C 实现。

> ⚠️ **本关新增"造bug高峰"**：§04 故意把左结合写成右结合 → `1-2-3` 算成 `2` 而不是 `-4` → 亲眼看到"语法错误改变语义"。

## 02.Lexer词法分析

```
┌─ 🎯 阶段 ① 目标 ────────────────────────────────┐
│ 把 "1+2*3" → [{NUM,1},{+},{NUM,2},{*},{NUM,3}]      │
└──────────────────────────────────────────────────┘
```

```c
typedef enum {
    TOK_NUM, TOK_PLUS, TOK_MINUS, TOK_STAR, TOK_SLASH,
    TOK_LPAREN, TOK_RPAREN, TOK_IDENT, TOK_ASSIGN, TOK_EOF
} TokenKind;

typedef struct {
    TokenKind kind;
    int       value;      // 数字值
    char      name[32];   // 标识符名
} Token;

// 词法分析器
typedef struct {
    const char *src;
    int         pos;
} Lexer;

Token lexer_next(Lexer *l) {
    // 跳过空白
    while (l->src[l->pos] == ' ' || l->src[l->pos] == '\t') l->pos++;

    char c = l->src[l->pos];
    if (c == '\0') return (Token){TOK_EOF};

    // 数字
    if (c >= '0' && c <= '9') {
        int v = 0;
        while (l->src[l->pos] >= '0' && l->src[l->pos] <= '9')
            v = v * 10 + (l->src[l->pos++] - '0');
        return (Token){TOK_NUM, v};
    }

    // 标识符
    if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z')) {
        Token t = {TOK_IDENT, 0};
        int i = 0;
        while ((l->src[l->pos] >= 'a' && l->src[l->pos] <= 'z') ||
               (l->src[l->pos] >= 'A' && l->src[l->pos] <= 'Z') ||
               (l->src[l->pos] >= '0' && l->src[l->pos] <= '9'))
            t.name[i++] = l->src[l->pos++];
        t.name[i] = '\0';
        return t;
    }

    // 单字符运算符
    l->pos++;
    switch (c) {
        case '+': return (Token){TOK_PLUS};
        case '-': return (Token){TOK_MINUS};
        case '*': return (Token){TOK_STAR};
        case '/': return (Token){TOK_SLASH};
        case '(': return (Token){TOK_LPAREN};
        case ')': return (Token){TOK_RPAREN};
        case '=': return (Token){TOK_ASSIGN};
    }
    return (Token){TOK_EOF};
}
```

🧪 测试：输入 `1+2*3` → 输出 Token 序列 `[1,+,2,*,3]` ✅

## 03.Parser语法分析

```
┌─ 🎯 阶段 ② 目标 ────────────────────────────────┐
│ Token流 → AST 树（递归下降 + 优先级爬升）           │
└──────────────────────────────────────────────────┘
```

### 3.1 AST节点

```c
typedef enum { N_NUM, N_BINOP, N_IDENT, N_ASSIGN } NodeKind;

typedef struct AstNode {
    NodeKind kind;
    int      value;       // N_NUM 用
    char     name[32];    // N_IDENT 用
    int      op;          // N_BINOP 用（TokenKind）
    struct AstNode *left;
    struct AstNode *right;
} AstNode;

AstNode* makeNum(int v) {
    AstNode *n = malloc(sizeof(AstNode));
    n->kind = N_NUM; n->value = v;
    n->left = n->right = NULL;
    return n;
}

AstNode* makeBinOp(int op, AstNode *l, AstNode *r) {
    AstNode *n = malloc(sizeof(AstNode));
    n->kind = N_BINOP; n->op = op;
    n->left = l; n->right = r;
    return n;
}
```

### 3.2 递归下降Parser

```c
typedef struct {
    Token *tokens;
    int    pos;
} Parser;

Token peek(Parser *p) { return p->tokens[p->pos]; }
Token next (Parser *p) { return p->tokens[p->pos++]; }

// 前向声明
AstNode* parseExpression(Parser *p);
AstNode* parsePrimary(Parser *p) {
    Token t = next(p);
    if (t.kind == TOK_NUM)    return makeNum(t.value);
    if (t.kind == TOK_IDENT)  {
        AstNode *n = malloc(sizeof(AstNode));
        n->kind = N_IDENT; strcpy(n->name, t.name);
        n->left = n->right = NULL;
        return n;
    }
    if (t.kind == TOK_LPAREN) {
        AstNode *inner = parseExpression(p);
        next(p);  // 消费 )
        return inner;
    }
    return NULL;
}

AstNode* parseMulDiv(Parser *p) {
    AstNode *l = parsePrimary(p);
    while (peek(p).kind == TOK_STAR || peek(p).kind == TOK_SLASH) {
        int op = next(p).kind;
        AstNode *r = parsePrimary(p);
        l = makeBinOp(op, l, r);
    }
    return l;
}

AstNode* parseExpression(Parser *p) {
    AstNode *l = parseMulDiv(p);
    while (peek(p).kind == TOK_PLUS || peek(p).kind == TOK_MINUS) {
        int op = next(p).kind;
        AstNode *r = parseMulDiv(p);  // ⭐ 调高优先级层
        l = makeBinOp(op, l, r);
    }
    return l;
}
```

🔑 **while + 高优先级层调用 = 左结合**。如果把 `parseMulDiv(p)` 换成 `parseExpression(p)`（递归调自己），就变成右结合——`1-2-3` 会算成 `1-(2-3)=2`。

## 04.Interpreter解释执行

```
┌─ ⚠️ 造BUG高峰：左结合 vs 右结合 ──────────────────┐
│ 1-2-3: 正确 = (1-2)-3 = -4    错误 = 1-(2-3) = 2    │
└──────────────────────────────────────────────────────┘
```

```c
#include <string.h>

// 变量表（最多 26 个，用字母索引）
int vars[26] = {0};  // vars['a'-'a'] = 值

int interpret(AstNode *node) {
    if (!node) return 0;
    switch (node->kind) {
        case N_NUM:   return node->value;
        case N_IDENT: return vars[node->name[0] - 'a'];
        case N_BINOP:
            switch (node->op) {
                case TOK_PLUS:  return interpret(node->left) + interpret(node->right);
                case TOK_MINUS: return interpret(node->left) - interpret(node->right);
                case TOK_STAR:  return interpret(node->left) * interpret(node->right);
                case TOK_SLASH: return interpret(node->left) / interpret(node->right);
            }
        case N_ASSIGN: {
            int v = interpret(node->right);
            vars[node->name[0] - 'a'] = v;
            return v;
        }
    }
    return 0;
}
```

🧪 测试：
```text
输入: 1 + 2 * 3      → 7    （优先级正确）
输入: 1 - 2 - 3      → -4   （左结合正确）
输入: x = 5; x * 2   → 10   （变量支持）
```

### 释放AST内存

```c
void freeAst(AstNode *node) {
    if (!node) return;
    freeAst(node->left);
    freeAst(node->right);
    free(node);
}
```

## 05.扩展：变量与函数

```c
// 解析赋值语句: IDENT '=' expr
AstNode* parseAssignment(Parser *p) {
    Token t = peek(p);
    if (t.kind == TOK_IDENT && p->tokens[p->pos + 1].kind == TOK_ASSIGN) {
        AstNode *n = malloc(sizeof(AstNode));
        n->kind = N_ASSIGN;
        strcpy(n->name, t.name);
        next(p); next(p);  // 消费 IDENT 和 =
        n->right = parseExpression(p);
        n->left = NULL;
        return n;
    }
    return parseExpression(p);
}
```

## 06.总结与C++对比

| 维度 | C语言实现 | C++实现 |
|------|-----------|----------|
| **Token** | `enum + struct` | `std::variant` 类型安全 |
| **AST** | `struct` + 手动 `malloc/free` | `shared_ptr<AstNode>` |
| **Parser** | 递归下降 | 递归下降 + Visitor 模式 |
| **变量表** | 固定数组 `vars[26]` | `unordered_map<string, int>` |
| **内存管理** | 手动 `freeAst` | RAII 自动析构 |
| **错误处理** | `return NULL` | throw 异常 |

### 延伸挑战

1. **加函数调用**：`fn add(a,b){return a+b}` → `add(1,2)` → `3`
2. **加字符串支持**：`"hello" + " world"` → `"hello world"`
3. **加 if/while 控制流**：`if(x>0){print x}`
4. **移植到 C++**：用 `variant` 替换 Token、用 `shared_ptr` 替换裸指针

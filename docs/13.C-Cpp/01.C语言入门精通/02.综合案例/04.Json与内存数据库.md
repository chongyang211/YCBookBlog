## 02.数据结构定义

```c
#define MAX_NAME      64
#define MAX_COLUMNS   10
#define MAX_ROWS      100
#define MAX_TABLES    10
#define MAX_STR       256

typedef enum { TYPE_INT, TYPE_FLOAT, TYPE_STRING } DataType;

typedef struct {
    char     name[MAX_NAME];
    DataType type;
} Column;

typedef struct {
    void *data[MAX_COLUMNS];  // ⭐ void* 泛型存储
} Row;

typedef struct {
    char    name[MAX_NAME];
    Column  columns[MAX_COLUMNS];
    int     colCount;
    Row     rows[MAX_ROWS];
    int     rowCount;
} Table;

typedef struct {
    Table tables[MAX_TABLES];
    int   tableCount;
} Database;
```

🔑 **`void*` 泛型存储**：每列的数据类型不同（int/float/string），统一用 `void*` 指针存储。取数据时根据 `Column.type` 做强制转换——这是 C 语言的"类型擦除"。

## 03.创建表

```c
Table *createTable(Database *db, const char *tableName,
                   Column *cols, int colCount) {
    if (db->tableCount >= MAX_TABLES) return NULL;
    Table *t = &db->tables[db->tableCount++];
    strcpy(t->name, tableName);
    t->colCount = colCount;
    t->rowCount = 0;
    for (int i = 0; i < colCount; i++) {
        t->columns[i] = cols[i];
    }
    printf(">>> 表 '%s' 创建成功！(%d 列)\n", tableName, colCount);
    return t;
}
```

## 04.增删改查

```c
void insertRow(Table *t, void **values) {
    if (t->rowCount >= MAX_ROWS) { printf("行已满！\n"); return; }
    Row *r = &t->rows[t->rowCount];
    for (int i = 0; i < t->colCount; i++) {
        switch (t->columns[i].type) {
            case TYPE_INT:
                r->data[i] = malloc(sizeof(int));
                *(int*)r->data[i] = *(int*)values[i];
                break;
            case TYPE_FLOAT:
                r->data[i] = malloc(sizeof(float));
                *(float*)r->data[i] = *(float*)values[i];
                break;
            case TYPE_STRING:
                r->data[i] = malloc(MAX_STR);
                strcpy((char*)r->data[i], (char*)values[i]);
                break;
        }
    }
    t->rowCount++;
}

void queryTable(Table *t) {
    // 打印表头
    for (int i = 0; i < t->colCount; i++)
        printf("%-20s", t->columns[i].name);
    printf("\n");
    // 打印数据
    for (int i = 0; i < t->rowCount; i++) {
        for (int j = 0; j < t->colCount; j++) {
            switch (t->columns[j].type) {
                case TYPE_INT:    printf("%-20d", *(int*)t->rows[i].data[j]); break;
                case TYPE_FLOAT:  printf("%-20.2f", *(float*)t->rows[i].data[j]); break;
                case TYPE_STRING: printf("%-20s", (char*)t->rows[i].data[j]); break;
            }
        }
        printf("\n");
    }
}
```

## 05.二进制持久化

```c
void saveDatabase(Database *db, const char *filename) {
    FILE *f = fopen(filename, "wb");
    fwrite(&db->tableCount, sizeof(int), 1, f);
    for (int i = 0; i < db->tableCount; i++) {
        Table *t = &db->tables[i];
        fwrite(t->name, sizeof(char), MAX_NAME, f);
        fwrite(&t->colCount, sizeof(int), 1, f);
        fwrite(t->columns, sizeof(Column), t->colCount, f);  // ⭐ 只写 colCount 个
        fwrite(&t->rowCount, sizeof(int), 1, f);
        for (int j = 0; j < t->rowCount; j++) {
            for (int k = 0; k < t->colCount; k++) {
                switch (t->columns[k].type) {
                    case TYPE_INT:   fwrite(t->rows[j].data[k], sizeof(int), 1, f); break;
                    case TYPE_FLOAT: fwrite(t->rows[j].data[k], sizeof(float), 1, f); break;
                    case TYPE_STRING: fwrite(t->rows[j].data[k], sizeof(char), MAX_STR, f); break;
                }
            }
        }
    }
    fclose(f);
    printf(">>> 数据库保存成功！\n");
}
```

## 06.类型安全与陷阱

```
┌─ ⚠️ 造BUG高峰 ────────────────────────────────────┐
│ 故意用错误的 DataType 读取 → 乱码/崩溃              │
└──────────────────────────────────────────────────────┘
```

```c
// ❌ 假设某列的 type 写成了 TYPE_STRING，实际存的是 int
// 读取时：
printf("%s", (char*)row->data[k]);  // 把 int 的 4 字节当字符串地址 → 崩溃！
```

✅ **修复**：**所有访问 `void* data` 的地方，必须先检查 `columns[k].type`，用 `switch` 分派。**

## 07.JSON输出

```c
void tableToJson(Table *t) {
    printf("{\n  \"table\": \"%s\",\n  \"columns\": [", t->name);
    for (int i = 0; i < t->colCount; i++) {
        printf("\"%s\"%s", t->columns[i].name,
               i < t->colCount - 1 ? "," : "");
    }
    printf("],\n  \"rows\": [\n");
    for (int i = 0; i < t->rowCount; i++) {
        printf("    {");
        for (int j = 0; j < t->colCount; j++) {
            printf("\"%s\":", t->columns[j].name);
            switch (t->columns[j].type) {
                case TYPE_INT:    printf("%d", *(int*)t->rows[i].data[j]); break;
                case TYPE_FLOAT:  printf("%.2f", *(float*)t->rows[i].data[j]); break;
                case TYPE_STRING: printf("\"%s\"", (char*)t->rows[i].data[j]); break;
            }
            if (j < t->colCount - 1) printf(",");
        }
        printf("}%s\n", i < t->rowCount - 1 ? "," : "");
    }
    printf("  ]\n}\n");
}
```

## 08.总结与C++对比

| 维度 | C语言实现 | C++实现 |
|------|-----------|----------|
| **泛型数据** | `void*` + `DataType` 枚举 | `template` / `std::variant` |
| **容器** | 定长数组 | `std::vector` |
| **类型安全** | 手动 switch 分派 | 编译期 `std::visit` |
| **JSON** | 手动 `printf` 拼接 | `nlohmann/json` 库 |
| **二进制** | `fwrite`/`fread` 逐字段 | `boost::serialization` |

**延伸挑战**：加索引（哈希表加速查找）、加 SQL 解析器、用 `variant` 重写 C++ 版。

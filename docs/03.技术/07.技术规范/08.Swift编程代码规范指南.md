## 01.命名规范

| 类型 | 规范 | 示例 |
|:-----|:-----|:-----|
| 类/结构体/枚举 | 大驼峰 | `UserProfile`, `HTTPClient` |
| 方法/属性 | 小驼峰 | `fetchUser()`, `userName` |
| 布尔属性 | `is/has/should` 前缀 | `isEmpty`, `hasData` |
| 协议 | 名词或 `able/ible/ing` | `Codable`, `Equatable` |
| 常量 | 小驼峰 | `maxRetryCount` |

```swift
// ✅ 方法名读起来像英文句子
func insert(_ element: Element, at index: Int)
func move(from source: Int, to destination: Int)

// ✅ 工厂方法参数是创建的核心
let instance = UIView(frame: .zero)          // 不是 makeView()
let date = Date(timeIntervalSinceNow: 3600)  // 不是 makeDate()

// ✅ 布尔属性读起来像断言
var isEmpty: Bool { return count == 0 }
```

## 02.代码格式

```swift
// ✅ 缩进：4 个空格
// ✅ 冒号：类型声明时右边有空格
class User {
    let name: String
    let age: Int
    private var data: [String: Any] = [:]
}

// ✅ 多行参数：左括号换行，参数对齐
func fetch(
    from url: URL,
    method: HTTPMethod = .get,
    headers: [String: String] = [:],
) async throws -> Data

// ✅ import 按字母序
import Foundation
import SwiftUI
import UIKit
```

## 03.可选值处理

```swift
// ✅ guard let 提前返回（减少嵌套）
guard let user = currentUser else { return }

// ✅ if let 链
if let name = user?.name, let email = user?.email {
    sendEmail(to: email, name: name)
}

// ✅ nil 合并
let displayName = user?.name ?? "未知用户"

// ✅ 可选链
let city = user?.address?.city?.name

// ✅ 可选 map/flatMap
let uppercased = user?.name.map { $0.uppercased() }

// ❌ 避免强制解包
let name = user!.name  // 除非确定非空
```

## 04.闭包与函数

```swift
// ✅ 尾随闭包
UIView.animate(withDuration: 0.3) {
    view.alpha = 0
} completion: { _ in
    view.removeFromSuperview()
}

// ✅ 捕获列表解决循环引用
class ViewController {
    func setup() {
        api.fetch { [weak self] result in
            guard let self else { return }
            self.updateUI(result)
        }
    }
}

// ✅ 单行闭包用 $0
let names = users.map { $0.name }
let active = users.filter { $0.isActive }

// ✅ 闭包参数类型可推断时省略
let sorted = users.sorted { $0.name < $1.name }
```

## 05.枚举与协议

```swift
// ✅ 枚举带关联值
enum Result<T, E: Error> {
    case success(T)
    case failure(E)
}

// ✅ 协议组合
typealias CodableEquatable = Codable & Equatable

// ✅ 扩展协议提供默认实现
protocol Reusable {
    static var reuseIdentifier: String { get }
}
extension Reusable {
    static var reuseIdentifier: String { String(describing: self) }
}

// ✅ 带 RawValue 的枚举
enum HTTPMethod: String {
    case get = "GET"
    case post = "POST"
}
```

## 06.async/await

```swift
// ✅ 结构化并发
func loadUser(id: String) async throws -> User {
    let profile = try await fetchProfile(id: id)
    let posts = try await fetchPosts(for: id)
    return User(profile: profile, posts: posts)
}

// ✅ 并行任务（async let）
func loadAll() async throws -> (User, [Post]) {
    async let user = fetchUser()
    async let posts = fetchPosts()
    return try await (user, posts)
}

// ✅ TaskGroup 动态并发
func fetchAll(ids: [String]) async throws -> [User] {
    try await withThrowingTaskGroup(of: User.self) { group in
        for id in ids { group.addTask { try await fetchUser(id: id) } }
        return try await group.reduce(into: []) { $0.append($1) }
    }
}

// ✅ MainActor 标注 UI 代码
@MainActor
class UserViewModel: ObservableObject {
    @Published var users: [User] = []
    func load() async { users = try await api.fetchUsers() }
}
```

## 07.属性包装器

```swift
// ✅ 常用包装器
struct ContentView: View {
    @State private var text = ""
    @ObservedObject var viewModel: UserViewModel
    var body: some View { TextField("输入", text: $text) }
}

// ✅ 自定义属性包装器
@propertyWrapper
struct Clamped<T: Comparable> {
    private var value: T
    let range: ClosedRange<T>
    var wrappedValue: T {
        get { value }
        set { value = min(max(newValue, range.lowerBound), range.upperBound) }
    }
    init(wrappedValue: T, _ range: ClosedRange<T>) {
        self.range = range
        self.value = min(max(wrappedValue, range.lowerBound), range.upperBound)
    }
}
// 使用：@Clamped(0...100) var score = 50
```

## 08.类与结构体

```swift
// ✅ 优先用 struct（值语义，线程安全）
struct UserProfile {
    let id: String
    var name: String
}

// ✅ 需要引用语义时用 class
class DataManager {
    private var cache: [String: Any] = [:]
    func store(_ value: Any, for key: String) {
        cache[key] = value
    }
}

// ✅ 访问控制：默认 internal，按需 public/private
public class APIClient {
    public func fetch() async throws -> Data { ... }
    private func buildURL() -> URL { ... }
}
```

## 09.常见反模式

| 反模式 | 问题 | 改进 |
|:-------|:-----|:-----|
| 强制解包 `!` | 崩溃风险 | `guard let` / `??` |
| 循环引用 | 内存泄漏 | `[weak self]` |
| `try!` 忽略错误 | 崩溃 | `try?` 或 `do-catch` |
| 巨型 ViewController | 难以维护 | MVVM 拆分 |
| `var` 过多 | 可变状态难跟踪 | 优先 `let` |

## 10.代码审查清单

```markdown
## 命名
- [ ] 类/结构体/枚举大驼峰，方法/属性小驼峰
- [ ] 布尔属性 is/has/should 前缀

## 安全
- [ ] 无强制解包（!），无 try!
- [ ] 闭包中 [weak self] 防止循环引用
- [ ] 可选值用 guard let / if let / ?? 处理

## 并发
- [ ] UI 操作标记 @MainActor
- [ ] async/await 正确使用，无阻塞调用

## 设计
- [ ] 优先 struct 而非 class
- [ ] 协议小而精
- [ ] 访问控制合理（private/fileprivate/internal/public）
```

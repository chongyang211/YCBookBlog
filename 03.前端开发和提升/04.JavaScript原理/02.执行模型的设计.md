# 专栏笔记总结大全


2.1 栈内存（Stack）
存储基本类型值（Number、String、Boolean、null、undefined、Symbol、BigInt）
按值访问，大小固定，自动分配释放
2.2 堆内存（Heap）
存储引用类型值（Object、Array、Function）
按引用访问，动态分配，需要垃圾回收
// 栈内存示例
let a = 10;        // 在栈中存储值 10
let b = a;         // 在栈中创建新值 10 的副本
b = 20;           // 修改副本，不影响原值
console.log(a);    // 10 - 原值不变

// 堆内存示例
let obj1 = { name: "Alice" };  // 对象存储在堆中，栈中存储引用地址
let obj2 = obj1;               // 复制引用地址，指向同一个堆对象
obj2.name = "Bob";            // 修改堆中的对象
console.log(obj1.name);        // "Bob" - 两个引用指向同一对象
2.3 垃圾回收机制
标记清除算法（Mark-and-Sweep）：
function createObjects() {
let tempObj = { data: "临时对象" };  // 函数执行完，tempObj 不可达，被标记为可回收
return { permanent: "永久对象" };
}

let permanentRef = createObjects();  // 只有返回的对象保持可达
引用计数问题：
// 循环引用导致内存泄漏
function createCycle() {
let objA = { name: "A" };
let objB = { name: "B" };

    objA.ref = objB;  // objA 引用 objB
    objB.ref = objA;  // objB 引用 objA - 循环引用！
    
    return "完成";
}

createCycle();  // objA 和 objB 互相引用，但函数外不可达，现代 GC 能处理

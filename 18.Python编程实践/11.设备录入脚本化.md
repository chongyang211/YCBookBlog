# XXX 转换成curl请求
def trans_curl(url: str, headers: dict, body: bytes, method: str) -> str:
    '''转换请求成curl
    
    :param url required: url
    :param headers required: headers
    :param body required: body
    :param method required: get/post
    
    :return: 返回curl请求字符串
    
    :raise exception_type: Description of the exception.
    '''
    if headers.get('x-zerus-env'):
        headers = f"-H 'Content-Type: application/json' -H 'x-zerus-env: {headers.get('x-zerus-env')}'"
    elif headers.get('Cookie'):
        headers = f"-H 'Content-Type: application/json' -H 'Cookie: {headers.get('Cookie')}'"
    elif headers.get('Authorization'):
        headers = f"-H 'Content-Type: application/json' -H 'Authorization: {headers.get('Authorization')}'"
    else:
        headers = "-H 'Content-Type: application/json'"
    if method == 'POST':
        body = f"--data-raw '{body.decode()}'"
        url = f"curl -X POST '{url}'"
        curl = url + ' ' + headers + ' ' + body
        return curl
    elif method == 'GET':
        url = f"curl '{url}'"
        curl = url + ' ' + headers
        return curl

# 获取openapi的token
def get_access_token():
    app_key = ""
    app_secret = ""
    ocode = ""
    host = "https://open.wecard.qq.com"
    access_token_url = host + "/cgi-bin/oauth2/token"
    access_token_data = {
        "app_key": app_key,
        "app_secret": app_secret,
        "grant_type": "client_credentials",  # 默认填client_credentials
        "scope": "base",  # 默认填 base
        "ocode": ocode
    }
    headers = {
        "Content-Type": "application/json",
        # "x-zerus-env": "vt-lqarlkjkvd77v"  # zerus环境，vt- 后面带上zerus环境id
    }
    access_token = requests.post(url=access_token_url, headers=headers, json=access_token_data)
    access_token = json.loads(access_token.text).get('access_token')
    return access_token

## 一、引言：为什么需要动态内存

### 1.1 静态分配的局限

```c
// 编译时必须确定数组大小
#define MAX_STUDENTS 100
struct Student students[MAX_STUDENTS];

// 问题：
// 1. 如果实际只有10个学生 → 浪费90个位置的内存
// 2. 如果有200个学生 → 数组不够用
// 3. 栈上的大数组可能导致栈溢出（默认栈8MB）
```

### 1.2 动态内存解决的问题

```c
// 运行时决定大小
int n;
printf("请输入学生人数: ");
scanf("%d", &n);

struct Student *students = malloc(n * sizeof(struct Student));
if (students == NULL) {
    fprintf(stderr, "内存分配失败\n");
    return 1;
}

// 使用...

// 不再需要时释放
free(students);
```

### 1.3 动态内存的三个核心问题

```
1. 分配: 如何高效地从操作系统获取内存？
2. 使用: 如何安全地读写分配的内存？
3. 释放: 如何正确地归还内存？如何避免泄漏？
```

## 二、malloc/calloc/realloc/free详解

### 2.1 malloc

```c
void *malloc(size_t size);

// 分配size字节的未初始化内存
// 返回值：成功返回指向内存的指针，失败返回NULL
// 内存内容：未定义（可能是旧数据）

int *p = malloc(10 * sizeof(int));
if (p == NULL) {
    perror("malloc");
    exit(EXIT_FAILURE);
}

// 推荐写法：使用sizeof(*p)而非sizeof(int)
int *q = malloc(10 * sizeof(*q));  // 更安全，类型变更不需修改
```

### 2.2 calloc

```c
void *calloc(size_t nmemb, size_t size);

// 分配nmemb个大小为size的元素，并初始化为0
// 等价于 malloc(nmemb * size) + memset(0)
// 但calloc还会检查 nmemb * size 的乘法溢出

int *arr = calloc(1000, sizeof(int));
// 所有元素自动初始化为0

// calloc vs malloc+memset 的区别：
// 1. calloc检查乘法溢出
// 2. calloc可能利用操作系统的零页优化（延迟初始化）
```

### 2.3 realloc

```c
void *realloc(void *ptr, size_t new_size);

// 调整已分配内存的大小
// 三种行为：
// ptr == NULL → 等同于 malloc(new_size)
// new_size == 0 → 行为由实现定义（不推荐使用）
// 否则 → 尝试扩展或重新分配

// 重要：realloc可能返回不同地址！
int *p = malloc(100 * sizeof(int));
// ... 使用p ...

// 错误用法：
// p = realloc(p, 200 * sizeof(int));  // 如果失败，p变成NULL，原内存泄漏！

// 正确用法：
int *temp = realloc(p, 200 * sizeof(int));
if (temp == NULL) {
    // realloc失败，p仍然有效
    free(p);
    return NULL;
}
p = temp;  // 成功，更新指针
```

### 2.4 free

```c
void free(void *ptr);

// 释放malloc/calloc/realloc分配的内存
// ptr == NULL → 无操作（安全）
// ptr指向非动态分配的内存 → 未定义行为
// 同一ptr free两次 → 未定义行为（double free）

int *p = malloc(sizeof(int));
free(p);
p = NULL;  // 好习惯：free后置NULL

// free(NULL) 是安全的，可以简化代码
void cleanup(int *a, int *b, int *c) {
    free(a);  // 即使a是NULL也安全
    free(b);
    free(c);
}
```

### 2.5 aligned_alloc（C11）

```c
void *aligned_alloc(size_t alignment, size_t size);

// 分配对齐到alignment的内存
// alignment必须是2的幂
// size必须是alignment的整数倍

double *data = aligned_alloc(64, 1024 * sizeof(double));
// data的地址是64字节对齐的（缓存行对齐）

// POSIX替代方案
void *ptr;
int ret = posix_memalign(&ptr, 64, 1024 * sizeof(double));
if (ret != 0) {
    // 失败
}
```

## 三、malloc的底层实现原理

### 3.1 系统调用层

```
应用程序
    ↓ malloc(size)
C运行时库（glibc ptmalloc2）
    ↓ 根据size选择策略
┌────────────────────────────────┐
│ size < 128KB    size >= 128KB  │
│     ↓                ↓        │
│   brk/sbrk       mmap        │
│  (调整堆顶)    (匿名映射)     │
└────────────────────────────────┘
    ↓
操作系统内核
    ↓ 分配虚拟页 + 建立页表映射
物理内存
```

### 3.2 brk/sbrk系统调用

```c
#include <unistd.h>

// brk设置堆顶地址
int brk(void *addr);

// sbrk增加/减少堆空间
void *sbrk(intptr_t increment);

// 示例
void *start = sbrk(0);     // 获取当前堆顶
printf("堆顶: %p\n", start);

sbrk(4096);                 // 扩展4KB
printf("新堆顶: %p\n", sbrk(0));

sbrk(-4096);                // 收缩4KB
printf("收缩后: %p\n", sbrk(0));
```

### 3.3 mmap分配大块内存

```c
#include <sys/mman.h>

// 大块内存使用mmap匿名映射
void *ptr = mmap(NULL, size,
                 PROT_READ | PROT_WRITE,
                 MAP_PRIVATE | MAP_ANONYMOUS,
                 -1, 0);

// 释放
munmap(ptr, size);

// 为什么大块用mmap？
// 1. mmap的内存可以独立归还系统（munmap）
// 2. brk的堆是连续的，中间有已分配块时无法收缩
// 3. mmap不产生碎片
```

### 3.4 分配策略的阈值

```
glibc ptmalloc2 的默认阈值：

分配大小             策略
< 64字节            Fast Bins（无锁、LIFO）
64 ~ 512字节        Small Bins（FIFO、合并）
512B ~ 128KB        Large Bins（按大小排序）
≥ 128KB             mmap直接映射

// 阈值可以通过mallopt调整
#include <malloc.h>
mallopt(M_MMAP_THRESHOLD, 256 * 1024);  // 调整mmap阈值为256KB
mallopt(M_TRIM_THRESHOLD, 512 * 1024);  // 调整brk收缩阈值
```

## 四、内存分配器的核心数据结构

### 4.1 Chunk结构

```c
// glibc中的 malloc_chunk（简化版）
struct malloc_chunk {
    size_t prev_size;    // 前一个chunk的大小（仅当前一个chunk空闲时有效）
    size_t size;         // 当前chunk的大小（包含头部，低3位用作标志）
    
    // 以下仅在chunk空闲时使用
    struct malloc_chunk *fd;  // 空闲链表前向指针
    struct malloc_chunk *bk;  // 空闲链表后向指针
};

// size字段的低3位标志：
// bit 0 (P): PREV_INUSE  — 前一个chunk是否在使用
// bit 1 (M): IS_MMAPPED  — 是否通过mmap分配
// bit 2 (A): NON_MAIN_ARENA — 是否属于非主arena

// 获取实际大小：
#define CHUNK_SIZE(p) ((p)->size & ~0x7)
```

### 4.2 已分配chunk和空闲chunk的布局

```
已分配的chunk:
+------------------+
| prev_size        |  ← 如果前一个chunk已分配，此区域可被前一个chunk使用
+------------------+
| size       |A|M|P|  ← 实际大小 + 标志位
+------------------+
|                  |
| 用户数据 (payload)|  ← malloc返回的指针指向这里
|                  |
+------------------+

空闲的chunk:
+------------------+
| prev_size        |
+------------------+
| size       |A|M|P|
+------------------+
| fd (前向指针)     |  ← 指向空闲链表中下一个chunk
+------------------+
| bk (后向指针)     |  ← 指向空闲链表中上一个chunk
+------------------+
| 未使用空间        |
+------------------+
```

### 4.3 Bins分类

```
Bins是管理空闲chunk的数据结构：

1. Fast Bins (10个)
   - 单链表，LIFO
   - chunk大小: 16, 24, 32, ..., 80字节
   - 不合并相邻空闲chunk（牺牲空间换速度）
   - 特点：分配和释放最快

2. Small Bins (62个)
   - 双链表，FIFO
   - chunk大小: 16 ~ 512字节（每个bin固定大小）
   - 相邻空闲chunk会合并

3. Large Bins (63个)
   - 双链表，按大小排序
   - chunk大小: > 512字节
   - 使用best-fit策略

4. Unsorted Bin (1个)
   - 双链表
   - 新释放的chunk先放入unsorted bin
   - 下次分配时再分类到small/large bins
   - 类似于"缓冲区"

5. Top Chunk
   - 堆顶的特殊chunk
   - 没有合适的空闲chunk时，从top chunk分割
```

### 4.4 Arena（多线程优化）

```
ptmalloc2为多线程设计了arena机制：

主arena:  使用brk管理，全局唯一
非主arena: 使用mmap分配，每个线程可以有独立的arena

目的：减少线程间的锁竞争

线程数 ≤ 8:  每个线程一个arena
线程数 > 8:  arena数量 = 8 * CPU核心数

分配流程：
1. 尝试获取本线程上次使用的arena
2. 如果该arena已被锁定，尝试其他arena
3. 如果所有arena都忙，创建新arena
```

## 五、free的底层原理

### 5.1 free的工作流程

```
free(ptr)
    │
    ├─ ptr == NULL → 直接返回
    │
    ├─ 根据ptr找到chunk头部（ptr前面固定偏移）
    │
    ├─ 检查chunk合法性
    │   ├─ size是否合理
    │   └─ 对齐是否正确
    │
    ├─ 是mmap分配的？ → munmap直接归还系统
    │
    └─ 否则进入bins管理
        ├─ 大小属于fast bins → 放入对应fast bin（不合并）
        │
        └─ 否则：
            ├─ 检查相邻的前一个chunk是否空闲 → 合并
            ├─ 检查相邻的后一个chunk是否空闲 → 合并
            ├─ 合并后的chunk放入unsorted bin
            └─ 如果合并后与top chunk相邻 → 合并到top chunk
                └─ 如果top chunk足够大 → brk收缩，归还系统
```

### 5.2 空闲chunk合并

```
合并前:
[Chunk A: 已分配] [Chunk B: 空闲] [Chunk C: 刚释放] [Chunk D: 空闲]

合并后:
[Chunk A: 已分配] [    Chunk B+C+D: 空闲（合并为一个大chunk）    ]

合并条件：
- 前一个chunk空闲：通过当前chunk的P标志位判断
  如果P=0，prev_size字段记录了前一个chunk的大小
- 后一个chunk空闲：检查后一个chunk的下一个chunk的P标志位
```

### 5.3 为什么free后内存不一定归还系统？

```c
// 实验
void *p1 = malloc(1000);
void *p2 = malloc(1000);
void *p3 = malloc(1000);

free(p1);  // p1的内存放入空闲链表，不归还系统
free(p3);  // p3的内存放入空闲链表，不归还系统
// 因为p2还在使用，堆不能收缩

free(p2);  // p2释放后，p1+p2+p3可能合并
// 如果合并后在堆顶且足够大，才能通过brk收缩归还系统

// 这就是为什么进程的RSS（常驻内存）可能不会立即降低
```

## 六、内存碎片问题

### 6.1 外部碎片

```c
// 模拟外部碎片
void *blocks[100];
for (int i = 0; i < 100; i++) {
    blocks[i] = malloc(100);  // 100个100字节的块
}

// 释放奇数位置
for (int i = 1; i < 100; i += 2) {
    free(blocks[i]);
    blocks[i] = NULL;
}

// 现在有50*100=5000字节空闲，但分散在50个不连续的块中
// 无法分配一个5000字节的连续块
void *big = malloc(5000);  // 可能需要向系统申请新内存
```

### 6.2 内部碎片

```c
// malloc通常会对齐到16字节
void *p = malloc(1);  // 请求1字节
// 实际分配了32字节（16字节chunk头 + 16字节最小payload）
// 内部碎片 = 32 - 1 - 16 = 15字节

void *q = malloc(17);  // 请求17字节
// 实际分配了48字节（16字节头 + 32字节payload，对齐到16）
// 内部碎片 = 48 - 17 - 16 = 15字节
```

### 6.3 碎片的解决方案

```
1. 内存池（Pool Allocator）
   - 预分配大块内存，按固定大小分割
   - 适合大量相同大小的对象

2. Slab Allocator
   - Linux内核使用的分配器
   - 为每种对象大小维护独立的缓存

3. 紧凑（Compaction）
   - 移动对象消除碎片
   - C语言中难以实现（指针不能自动更新）
   - 需要间接引用（句柄机制）

4. 合适的分配策略
   - First-fit: 使用第一个足够大的空闲块
   - Best-fit: 使用最小的足够大的空闲块
   - Worst-fit: 使用最大的空闲块
```

## 七、内存泄漏的类型与检测

### 7.1 内存泄漏的分类

```c
// 1. 直接泄漏：分配后完全丢失引用
void direct_leak() {
    int *p = malloc(100);
    p = NULL;  // 指针覆盖，无法再free
}

// 2. 间接泄漏：通过泄漏的内存间接引用的内存
void indirect_leak() {
    struct Node { int data; struct Node *next; };
    struct Node *head = malloc(sizeof(struct Node));
    head->next = malloc(sizeof(struct Node));
    head->next->next = NULL;
    
    head = NULL;  // 泄漏head和head->next
}

// 3. 可能的泄漏（still reachable）：程序结束时仍有指针指向
int *global_ptr;
void possible_leak() {
    global_ptr = malloc(100);
    // 程序结束时没有free，但global_ptr仍然有效
    // 操作系统会回收，但在长期运行的服务中是问题
}

// 4. 增长型泄漏：泄漏速率随时间增长
void growing_leak() {
    while (1) {
        void *p = malloc(1024);
        // 忘记free，内存持续增长
        sleep(1);
    }
}
```

### 7.2 常见的泄漏场景

```c
// 1. 提前return忘记释放
int process(const char *input) {
    char *buffer = malloc(1024);
    
    if (input == NULL) {
        return -1;  // 泄漏！忘记free(buffer)
    }
    
    // ... 处理 ...
    free(buffer);
    return 0;
}

// 修复方案
int process_fixed(const char *input) {
    char *buffer = malloc(1024);
    int result = 0;
    
    if (input == NULL) {
        result = -1;
        goto cleanup;  // 使用goto统一清理
    }
    
    // ... 处理 ...

cleanup:
    free(buffer);
    return result;
}

// 2. 异常路径未释放
void complex_function() {
    FILE *f = fopen("data.txt", "r");
    char *data = malloc(1024);
    int *results = malloc(100 * sizeof(int));
    
    if (f == NULL) { free(data); free(results); return; }
    if (data == NULL) { fclose(f); free(results); return; }
    // 错误处理越来越复杂...
    
    // 更好的写法：使用goto集中清理
}

// 3. realloc失败导致泄漏
void realloc_leak() {
    int *p = malloc(100);
    p = realloc(p, 1000000000000);  // 可能失败
    // 如果失败，p=NULL，原来的100字节泄漏！
}
```

## 八、常见内存错误大全

### 8.1 错误类型总结

```c
// 1. 使用未初始化的内存
int *p = malloc(sizeof(int));
if (*p > 0) { ... }  // *p的值不确定！

// 2. 访问已释放的内存（Use-After-Free）
int *p = malloc(sizeof(int));
*p = 42;
free(p);
printf("%d\n", *p);  // 未定义行为！

// 3. 双重释放（Double Free）
int *p = malloc(sizeof(int));
free(p);
free(p);  // 第二次free → 未定义行为，可能导致堆损坏

// 4. 释放非动态分配的内存
int a = 10;
free(&a);  // 未定义行为！a在栈上

// 5. 缓冲区溢出（堆）
int *p = malloc(10 * sizeof(int));
p[10] = 42;  // 越界写入！覆盖相邻chunk的元数据

// 6. 缓冲区下溢
int *p = malloc(10 * sizeof(int));
p[-1] = 42;  // 覆盖chunk头部！

// 7. 内存泄漏
for (int i = 0; i < 1000; i++) {
    malloc(1024);  // 从未free
}

// 8. realloc后使用旧指针
int *old = malloc(100);
int *new_ptr = realloc(old, 200);
old[0] = 42;  // 如果realloc重新分配了内存，old已失效！

// 9. 偏移后free
int *p = malloc(100);
p += 5;
free(p);  // 错误！必须free原始指针

// 10. size_t溢出
size_t n = UINT_MAX;
int *p = malloc(n * sizeof(int));  // 乘法溢出！
```

### 9.2 检测示例

```c
// test.c
#include <stdlib.h>
#include <string.h>

void leak_demo() {
    int *p = malloc(100 * sizeof(int));
    // 忘记free → 内存泄漏
}

void uninit_demo() {
    int *p = malloc(sizeof(int));
    if (*p > 0) {  // 读取未初始化值
        printf("positive\n");
    }
    free(p);
}

void overflow_demo() {
    int *p = malloc(10 * sizeof(int));
    p[10] = 42;  // 堆缓冲区溢出

void use_after_free_demo() {
    int *p = malloc(sizeof(int));
    *p = 42;
    free(p);
    int x = *p;  // 使用已释放的内存
}

int main() {
    leak_demo();
    uninit_demo();
    overflow_demo();
    use_after_free_demo();
    return 0;
}
```

## 十、AddressSanitizer实战

### 10.1 ASan的使用

```bash
# 编译时启用ASan
gcc -fsanitize=address -g -O1 -fno-omit-frame-pointer -o test test.c

# 运行即可检测
./test
```

### 10.2 ASan的工作原理

```
ASan使用"影子内存"（Shadow Memory）技术：

应用内存    影子内存
8字节   →   1字节

影子字节的含义：
  0:     所有8字节可访问
  1-7:   前k字节可访问
  负值:  不可访问
    -1 (0xFF): 堆缓冲区的RedZone
    -2 (0xFE): 栈缓冲区的RedZone
    -3 (0xFD): 全局缓冲区的RedZone
    -4 (0xFC): 已释放的内存

内存布局（启用ASan后）：
[RedZone][用户数据][RedZone]
  ↑ 不可访问     ↑ 不可访问
  任何访问RedZone的操作都会被检测到

性能开销：约2x（比Valgrind的20x快很多）
内存开销：约3x
```

### 10.3 ASan可检测的错误

```
1. 堆缓冲区溢出/下溢
2. 栈缓冲区溢出
3. 全局缓冲区溢出
4. Use-After-Free
5. Use-After-Return（需要额外flag）
6. Double-Free
7. 内存泄漏（需要启用LeakSanitizer）
```

## 十一、自定义内存分配器

### 11.1 简单的分配器实现

```c
#include <stddef.h>
#include <sys/mman.h>

// 简单的堆分配器：使用空闲链表
typedef struct Block {
    size_t size;
    struct Block *next;
    int free;
} Block;

#define BLOCK_SIZE sizeof(Block)
#define ALIGN(size) (((size) + 7) & ~7)  // 8字节对齐

static Block *free_list = NULL;

// 查找空闲块（首次适配）
Block *find_free_block(size_t size) {
    Block *current = free_list;
    while (current) {
        if (current->free && current->size >= size) {
            return current;
        }
        current = current->next;
    }
    return NULL;
}

// 向系统请求内存
Block *request_space(size_t size) {
    Block *block = sbrk(0);
    void *request = sbrk(BLOCK_SIZE + size);
    if (request == (void*)-1) {
        return NULL;
    }
    block->size = size;
    block->next = NULL;
    block->free = 0;
    
    if (free_list == NULL) {
        free_list = block;
    }
    return block;
}

void *my_malloc(size_t size) {
    if (size == 0) return NULL;
    size = ALIGN(size);
    
    Block *block = find_free_block(size);
    if (block) {
        block->free = 0;
        return (void*)(block + 1);
    }
    
    block = request_space(size);
    if (!block) return NULL;
    
    return (void*)(block + 1);
}

void my_free(void *ptr) {
    if (!ptr) return;
    Block *block = (Block*)ptr - 1;
    block->free = 1;
    // 简化版：不做合并
}
```

### 11.2 添加分裂与合并

```c
// 如果空闲块比请求大得多，分裂为两块
void split_block(Block *block, size_t size) {
    if (block->size >= size + BLOCK_SIZE + 8) {
        Block *new_block = (Block*)((char*)(block + 1) + size);
        new_block->size = block->size - size - BLOCK_SIZE;
        new_block->next = block->next;
        new_block->free = 1;
        
        block->size = size;
        block->next = new_block;
    }
}

// 合并相邻的空闲块
void coalesce() {
    Block *current = free_list;
    while (current && current->next) {
        if (current->free && current->next->free) {
            current->size += BLOCK_SIZE + current->next->size;
            current->next = current->next->next;
        } else {
            current = current->next;
        }
    }
}
```

## 十二、内存池技术

### 12.1 固定大小内存池

```c
#define POOL_BLOCK_SIZE 64
#define POOL_CAPACITY 1024

typedef struct MemPool {
    char memory[POOL_BLOCK_SIZE * POOL_CAPACITY];
    int free_stack[POOL_CAPACITY];
    int top;
} MemPool;

void pool_init(MemPool *pool) {
    pool->top = POOL_CAPACITY;
    for (int i = 0; i < POOL_CAPACITY; i++) {
        pool->free_stack[i] = i;
    }
}

void *pool_alloc(MemPool *pool) {
    if (pool->top == 0) return NULL;
    int index = pool->free_stack[--pool->top];
    return &pool->memory[index * POOL_BLOCK_SIZE];
}

void pool_free(MemPool *pool, void *ptr) {
    int index = ((char*)ptr - pool->memory) / POOL_BLOCK_SIZE;
    pool->free_stack[pool->top++] = index;
}

// 使用
MemPool pool;
pool_init(&pool);

void *p1 = pool_alloc(&pool);  // O(1) 分配
void *p2 = pool_alloc(&pool);  // O(1) 分配
pool_free(&pool, p1);          // O(1) 释放
```

### 12.2 对象池

```c
typedef struct {
    int id;
    char name[32];
    float score;
} Student;

typedef struct {
    Student students[1000];
    int used[1000];  // 0=空闲, 1=已使用
    int count;
} StudentPool;

Student *student_alloc(StudentPool *pool) {
    for (int i = 0; i < 1000; i++) {
        if (!pool->used[i]) {
            pool->used[i] = 1;
            pool->count++;
            memset(&pool->students[i], 0, sizeof(Student));
            return &pool->students[i];
        }
    }
    return NULL;  // 池满
}

void student_free(StudentPool *pool, Student *s) {
    int index = s - pool->students;
    if (index >= 0 && index < 1000) {
        pool->used[index] = 0;
        pool->count--;
    }
}
```

## 十三、智能指针思想在C中的实现

### 13.1 引用计数

```c
typedef struct {
    void *ptr;
    size_t size;
    int ref_count;
    void (*destructor)(void*);
} RefCounted;

RefCounted *rc_create(size_t size, void (*dtor)(void*)) {
    RefCounted *rc = malloc(sizeof(RefCounted));
    rc->ptr = malloc(size);
    rc->size = size;
    rc->ref_count = 1;
    rc->destructor = dtor;
    return rc;
}

void rc_retain(RefCounted *rc) {
    if (rc) rc->ref_count++;
}

void rc_release(RefCounted *rc) {
    if (rc && --rc->ref_count == 0) {
        if (rc->destructor) {
            rc->destructor(rc->ptr);
        }
        free(rc->ptr);
        free(rc);
    }
}

// 使用
void string_destructor(void *p) {
    printf("销毁字符串: %s\n", (char*)p);
}

RefCounted *str = rc_create(100, string_destructor);
strcpy(str->ptr, "Hello, RefCount!");

rc_retain(str);   // ref_count = 2
rc_release(str);   // ref_count = 1
rc_release(str);   // ref_count = 0 → 自动释放
```

```c
int process_data() {

    char *buffer = NULL;
    int *results = NULL;
    int ret = -1;

    if (!buffer) goto cleanup;
    
    results = malloc(100 * sizeof(int));
    if (!results) goto cleanup;

    free(buffer);

    return ret;
}
```

### 14.2 调试辅助工具

```c
// 包装malloc/free，添加追踪
#ifdef DEBUG

typedef struct AllocInfo {
    void *ptr;
    size_t size;
    const char *file;
    int line;
    struct AllocInfo *next;
} AllocInfo;

static AllocInfo *alloc_list = NULL;
static size_t total_allocated = 0;

void *debug_malloc(size_t size, const char *file, int line) {
    void *ptr = malloc(size);
    if (ptr) {
        AllocInfo *info = malloc(sizeof(AllocInfo));
        info->ptr = ptr;
        info->size = size;
        info->file = file;
        info->line = line;
        info->next = alloc_list;
        alloc_list = info;
        total_allocated += size;
    }
    return ptr;
}

void debug_free(void *ptr, const char *file, int line) {
    if (!ptr) return;
    AllocInfo **prev = &alloc_list;
    AllocInfo *curr = alloc_list;
    while (curr) {
        if (curr->ptr == ptr) {
            total_allocated -= curr->size;
            *prev = curr->next;
            free(curr);
            free(ptr);
            return;
        }
        prev = &curr->next;
        curr = curr->next;
    }
    fprintf(stderr, "WARNING: free未知指针 %p at %s:%d\n", ptr, file, line);
}

void debug_report() {
    AllocInfo *curr = alloc_list;
    if (!curr) {
        printf("无内存泄漏\n");
        return;
    }
    printf("=== 内存泄漏报告 ===\n");
    while (curr) {
        printf("泄漏 %zu 字节 at %s:%d (ptr=%p)\n",
               curr->size, curr->file, curr->line, curr->ptr);
        curr = curr->next;
    }
    printf("总计泄漏: %zu 字节\n", total_allocated);
}

#define malloc(size) debug_malloc(size, __FILE__, __LINE__)
#define free(ptr) debug_free(ptr, __FILE__, __LINE__)

#endif
```

## 十五、总结与面试高频问题

### 15.1 核心知识总结

```
动态内存管理全景图：

系统层:
  brk/sbrk → 管理堆区（连续内存）
  mmap/munmap → 管理内存映射（独立区域）

分配器层:
  ptmalloc2 → Bins管理空闲chunk
  ├── Fast Bins: 小内存快速分配
  ├── Small/Large Bins: 中大内存管理
  ├── Unsorted Bin: 释放缓冲
  └── Arena: 多线程优化

应用层:
  malloc/calloc/realloc/free → 用户接口
  内存池 → 特殊场景优化
  引用计数 → 生命周期管理

安全检测:
  Valgrind → 全面但慢（20x）
  ASan → 快速有效（2x）
  自定义wrapper → 轻量追踪
```

### 15.2 面试高频问题

**Q1: malloc(0)返回什么？**
```
C标准：实现定义。glibc返回一个有效的非NULL指针（最小分配单元），可以free但不能解引用。
```

**Q2: free(NULL)会怎样？**
```
C标准保证：free(NULL)是安全的空操作。
```

**Q3: malloc和new的区别？**
```
malloc: C函数，返回void*，不调用构造函数，需要手动指定大小
new:    C++运算符，返回类型指针，调用构造函数，自动计算大小
```

**Q4: 如何检测内存泄漏？**
```
1. Valgrind（Linux）
2. AddressSanitizer（跨平台）
3. Visual Studio CRT调试（Windows）
4. 自定义malloc/free包装器
5. 静态分析工具（cppcheck等）
```

**Q5: 内存池的优点？**
```
1. 分配/释放O(1)（无需搜索空闲链表）
2. 无碎片（固定大小块）
3. 可预测的内存使用
4. 缓存友好（数据连续）
5. 无系统调用开销
```

虚函数（Virtual Function）和纯虚函数（Pure Virtual Function）是C++中实现多态的重要机制。它们的关键区别在于：

1. 
2. 纯虚函数：基类中不提供实现（即没有函数体），派生类必须重写该函数，否则派生类也会成为抽象类，无法实例化。

以下是虚函数和纯虚函数的示例：

```cpp
class Base{
public:
    virtual void virtualFunction() {
        // 虚函数的默认实现
        std::cout << "Base virtual function" << std::endl;
    }
    virtual void pureVirtualFunction() = 0;
};

class Derived : public Base {
public:
    void virtualFunction() override {
        // 重写虚函数
        std::cout << "Derived virtual function" << std::endl;
    }
    void pureVirtualFunction() override {
        // 实现纯虚函数
        std::cout << "Derived pure virtual function" << std::endl;
    }
};

int main() {
    Derived derived;
    derived.pureVirtualFunction();
    derived.virtualFunction();
    //new 创建对象
    Derived *derived2 = new Derived();
    derived2->pureVirtualFunction();
    derived2->virtualFunction();
    return 0;
}
```

这种设计允许我们定义接口（纯虚函数）和提供默认实现（虚函数），从而灵活地设计类层次结构。

**一句话概括**：**虚函数是"可以重写"的函数（有默认实现），纯虚函数是"必须重写"的函数（无实现，定义接口规范）。**

| 特性 | 虚函数 (Virtual Function) | 纯虚函数 (Pure Virtual Function) |
|------|--------------------------|---------------------------------|
| **语法** | `virtual void func();` | `virtual void func() = 0;` |
| **实现** | 基类中**有**函数体 | 基类中**无**函数体（=0） |
| **重写要求** | 派生类**可选**重写 | 派生类**必须**重写 |
| **类性质** | 可以是具体类 | 使类成为**抽象类** |
| **实例化** | 可以创建对象 | **不能**创建抽象类对象 |

## 12.3 多态和析构函数

### 12.3.1 多态中析构

多态使用时，如果子类中有属性开辟到堆区，那么父类指针在释放时无法调用到子类的析构代码。然后来看下，下面的代码。

```
#include <iostream>
#include <cstring>
using namespace std;

class Animal {
public:
    Animal() {
        cout << "Animal构造函数" << endl;
    }
    
    // ❌ 问题：析构函数不是虚函数！
    ~Animal() {
        cout << "Animal析构函数" << endl;
    }
};

class Dog : public Animal {
private:
    char* name;  // 在堆区分配内存的属性
public:
    Dog(const char* dogName) {
        cout << "Dog构造函数" << endl;
        name = new char[strlen(dogName) + 1];
        strcpy(name, dogName);
        cout << "创建狗狗: " << name << endl;
    }
    
    ~Dog() {
        cout << "Dog析构函数，释放: " << name << endl;
        delete[] name;  // 释放堆内存
        name = nullptr;
    }
};

class Cat : public Animal {
private:
    char* color;  // 在堆区分配内存的属性
public:
    Cat(const char* catColor) {
        cout << "Cat构造函数" << endl;
        color = new char[strlen(catColor) + 1];
        strcpy(color, catColor);
        cout << "创建猫咪，颜色: " << color << endl;
    }
    
    ~Cat() {
        cout << "Cat析构函数，释放颜色: " << color << endl;
        delete[] color;  // 释放堆内存
        color = nullptr;
    }
};

// 测试函数：展示内存泄漏问题
void testMemoryLeak() {
    cout << "=== 测试开始：父类指针指向子类对象 ===" << endl;
    // 使用父类指针指向子类对象（多态）
    Animal* animal1 = new Dog("旺财");
    Animal* animal2 = new Cat("白色");
    cout << "=== 准备释放对象 ===" << endl;
    // 释放内存 - 这里会出现问题！
    delete animal1;  // ❌ 只调用Animal的析构函数，Dog的析构函数没调用！
    delete animal2;  // ❌ 只调用Animal的析构函数，Cat的析构函数没调用！
    cout << "=== 测试结束 ===" << endl << endl;
}

// 对比测试：直接使用子类指针
void testNoMemoryLeak() {
    cout << "=== 对比测试：直接使用子类指针 ===" << endl;
    Dog* dog = new Dog("小黑");
    Cat* cat = new Cat("黄色");
    cout << "=== 准备释放对象 ===" << endl;
    delete dog;  // ✅ 正确调用Dog的析构函数
    delete cat;  // ✅ 正确调用Cat的析构函数
    cout << "=== 对比测试结束 ===" << endl << endl;
}

int main() {
    // 运行有问题的测试
    testMemoryLeak();
    // 运行正确的对比测试
    testNoMemoryLeak();
    return 0;
}
```

运行结果如下所示：遇到的问题是，使用多态时，如果基类的析构函数不是虚函数，那么通过基类指针删除派生类对象时，只会调用基类的析构函数，而不会调用派生类的析构函数，从而导致派生类中分配的资源（如堆内存）泄漏。

### 12.3.2 多态中虚析构

从输出可以看到：Dog和Cat的析构函数都没有被调用，堆内存name和color没有被释放产生了内存泄漏。

解决方式：将父类中的析构函数改为虚析构或者纯虚析构。

虚析构语法：virtual ~类名(){}

```cpp
class Animal {
public:
    Animal() {
        cout << "Animal构造函数" << endl;
    }
    
    // ✅ 解决方案：将析构函数声明为虚函数
    virtual ~Animal() {
        cout << "Animal析构函数" << endl;
    }
};

class Dog : public Animal {
private:
    char* name;
public:
    Dog(const char* dogName) {
        cout << "Dog构造函数" << endl;
        name = new char[strlen(dogName) + 1];
        strcpy(name, dogName);
        cout << "创建狗狗: " << name << endl;
    }
    
    virtual ~Dog() override {
        cout << "Dog析构函数，释放: " << name << endl;
        delete[] name;
        name = nullptr;
    }
};

class Cat : public Animal {
private:
    char* color;
public:
    Cat(const char* catColor) {
        cout << "Cat构造函数" << endl;
        color = new char[strlen(catColor) + 1];
        strcpy(color, catColor);
        cout << "创建猫咪，颜色: " << color << endl;
    }
    
    virtual ~Cat() override {
        cout << "Cat析构函数，释放颜色: " << color << endl;
        delete[] color;
        color = nullptr;
    }
};

// 修复后的测试函数
void testFixedMemory() {
    cout << "=== 修复后的测试 ===" << endl;
    Animal* animals[3];
    animals[0] = new Dog("旺财");
    animals[1] = new Cat("白色");
    animals[2] = new Dog("来福");
    
    cout << "=== 准备释放对象数组 ===" << endl;
    // 现在能正确调用所有析构函数
    for (int i = 0; i < 3; i++) {
        delete animals[i];  // ✅ 正确调用子类的析构函数
    }
    cout << "=== 修复测试结束 ===" << endl;
}

int main() {
    testFixedMemory();
    return 0;
}
```

```text

```

总结：如果一个类可能被继承，并且会通过基类指针来删除派生类对象，为了调用派生类析构函数，那么基类必须要有虚析构函数。

1. 虚析构或纯虚析构就是用来解决通过父类指针释放子类对象
2. 如果子类中没有堆区数据，可以不写为虚析构或纯虚析构
3. 拥有纯虚析构函数的类也属于抽象类

### 12.3.3 多态中纯虚析构

虚析构和纯虚析构共性：

1. 可以解决父类指针释放子类对象

## 12.4 动态绑定

### 12.4.1 理解动态绑定

动态绑定是多态的实现机制，它允许在运行时根据对象的实际类型调用相应的函数。

### 12.4.2 动态绑定案例

通过利用虚函数实现动态绑定，可以在运行时根据对象的实际类型来调用相应的函数，实现多态的效果。

这提供了更灵活和可扩展的代码结构，使得程序能够适应不同类型的对象并执行相应的操作。

```cpp
class Shape {
public:
    virtual void draw() {
        // 基类虚函数的默认实现
        cout << "Drawing a shape." << endl;
    }
};

class Circle : public Shape {
public:
    void draw() override {
        // 派生类虚函数的实现
        cout << "Drawing a circle." << endl;
    }
};

class Rectangle : public Shape {
public:
    void draw() override {
        // 派生类虚函数的实现
        cout << "Drawing a rectangle." << endl;
    }
};

int main() {
    Shape *shape;
    Circle circle;
    Rectangle rectangle;

    //示例中，基类Shape中的draw()函数被声明为虚函数。派生类Circle和Rectangle分别重写了基类的虚函数。
    //由于draw()函数是虚函数，因此在运行时会根据实际对象的类型来动态绑定，选择调用相应的派生类的函数实现。

    // 动态绑定，调用Circle类的draw()函数
    shape = &circle;
    shape->draw();

    // 动态绑定，调用Rectangle类的draw()函数
    shape = &rectangle;
    shape->draw();
    return 0;
}
```

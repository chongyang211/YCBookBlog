## 1.什么是状态机

### 1.1 一句话概括

状态机（State Machine）是一种数学模型，用于描述系统在不同状态之间的转换行为。它是一种抽象的计算模型，能够精确描述系统在任意时刻的状态以及状态之间的转换条件。

### 1.2 状态机定义

状态机是一种数学模型，用于描述系统在不同状态之间的行为。它由以下核心组件构成：

1. **状态（State）**： 系统在某一时刻的特定条件或模式。例如，灯的“开”和“关”状态。
2. **事件（Event）**： 触发状态转换的外部输入或条件。例如，按下开关按钮。
3. **转换（Transition）**：从一个状态到另一个状态的变化。例如，灯从“关”状态转换到“开”状态。
4. **动作（Action）**： 在状态转换时执行的操作。例如，点亮灯泡。

### 1.3 状态机类型

状态机主要分为两类：

1. **有限状态机（Finite State Machine, FSM）**： 系统状态的数量是有限的。例如，交通信号灯的状态（红、黄、绿）。
2. **无限状态机（Infinite State Machine）**：系统状态的数量是无限的。例如，计数器的状态（理论上可以无限增加）。

## 2.来看一个案例

### 2.1 业务需求分析

场景：订单有多个状态，包括：待支付、已支付、已发货、已收货、已完成、已取消。

订单状态转换规则：

```text
待支付 -> 已支付（支付）
已支付 -> 已发货（发货）
已发货 -> 已收货（确认收货）
已收货 -> 已完成（完成订单）
待支付 -> 已取消（取消订单）
已支付 -> 已取消（取消订单，需退款）
```

```java
import java.util.HashMap;
import java.util.Map;

/**
 * 不使用状态机的订单服务实现

 */
public class OrderServiceWithoutStateMachine {

    public enum OrderStatus {

        PAID,          // 已支付
        SHIPPED,       // 已发货
        RECEIVED,      // 已收货
        COMPLETED,     // 已完成
        CANCELLED,     // 已取消
        AFTER_SALES    // 售后处理
    }
    
    // 订单实体

        private String orderId;
        private OrderStatus status;
        private double amount;
        private LocalDateTime createTime;
        private String shippingInfo;
        private boolean isRefunded;
        
        public Order(String orderId, double amount) {
            this.orderId = orderId;
            this.amount = amount;
            this.status = OrderStatus.PENDING;
            this.createTime = LocalDateTime.now();
        }
        
        // 省略getter/setter...
    }
    
    // 订单服务
    public class OrderService {
    
        private Map<String, Order> orderStore = new HashMap<>();
        
        /**
         * 支付订单 - 充满条件判断的糟糕代码
         */
        public boolean payOrder(String orderId, double amount) {
            Order order = orderStore.get(orderId);
            if (order == null) {
                System.out.println("订单不存在");
                return false;
            }
            
            // 检查当前状态
            if (order.getStatus() != OrderStatus.PENDING) {
                System.out.println("订单状态不是待支付，无法支付");
                return false;
            }
            
            // 检查支付金额
            if (amount < order.getAmount()) {
                System.out.println("支付金额不足");
                return false;
            }
            
            // 模拟库存检查
            if (!checkInventory(orderId)) {
                System.out.println("库存不足");
                return false;
            }
            
            // 执行支付逻辑
            boolean paymentSuccess = processPayment(orderId, amount);
            if (!paymentSuccess) {
                System.out.println("支付失败");
                return false;
            }
            
            // 更新状态
            order.setStatus(OrderStatus.PAID);
            System.out.println("订单支付成功，状态变为已支付");
            
            // 发送支付成功通知
            sendPaymentNotification(orderId);
            
            return true;
        }
        
        /**
         * 发货订单 - 更多的条件判断
         */
        public boolean shipOrder(String orderId, String shippingInfo) {
            Order order = orderStore.get(orderId);
            if (order == null) {
                System.out.println("订单不存在");
                return false;
            }
            
            // 检查当前状态
            if (order.getStatus() != OrderStatus.PAID) {
                System.out.println("订单状态不是已支付，无法发货");
                return false;
            }
            
            // 检查物流信息
            if (shippingInfo == null || shippingInfo.trim().isEmpty()) {
                System.out.println("物流信息不能为空");
                return false;
            }
            
            // 检查收货地址
            if (!validateShippingAddress(orderId)) {
                System.out.println("收货地址验证失败");
                return false;
            }
            
            // 执行发货逻辑
            boolean shippingSuccess = processShipping(orderId, shippingInfo);
            if (!shippingSuccess) {
                System.out.println("发货处理失败");
                return false;
            }
            
            // 更新状态
            order.setStatus(OrderStatus.SHIPPED);
            order.setShippingInfo(shippingInfo);
            System.out.println("订单发货成功，状态变为已发货");
            
            // 发送发货通知
            sendShippingNotification(orderId);
            
            return true;
        }
        
        /**
         * 确认收货 - 重复的条件判断模式
         */
        public boolean confirmReceipt(String orderId, String receiptCode) {
            Order order = orderStore.get(orderId);
            if (order == null) {
                System.out.println("订单不存在");
                return false;
            }
            
            // 检查当前状态
            if (order.getStatus() != OrderStatus.SHIPPED) {
                System.out.println("订单状态不是已发货，无法确认收货");
                return false;
            }
            
            // 验证签收码
            if (!validateReceiptCode(orderId, receiptCode)) {
                System.out.println("签收码验证失败");
                return false;
            }
            
            // 更新状态
            order.setStatus(OrderStatus.RECEIVED);
            System.out.println("订单确认收货成功，状态变为已收货");
            
            // 自动确认倒计时开始
            startAutoConfirmTimer(orderId);
            
            return true;
        }
        
        /**
         * 取消订单 - 最复杂的方法，包含多种取消逻辑
         */
        public boolean cancelOrder(String orderId, String cancelReason, boolean needRefund) {
            Order order = orderStore.get(orderId);
            if (order == null) {
                System.out.println("订单不存在");
                return false;
            }
            
            OrderStatus currentStatus = order.getStatus();
            
            // 根据当前状态执行不同的取消逻辑
            if (currentStatus == OrderStatus.PENDING) {
                // 待支付取消，直接取消
                order.setStatus(OrderStatus.CANCELLED);
                System.out.println("订单已取消（待支付状态）");
                
            } else if (currentStatus == OrderStatus.PAID) {
                // 已支付取消，需要退款
                if (needRefund) {
                    boolean refundSuccess = processRefund(orderId, order.getAmount());
                    if (!refundSuccess) {
                        System.out.println("退款处理失败");
                        return false;
                    }
                    order.setRefunded(true);
                }
                order.setStatus(OrderStatus.CANCELLED);
                System.out.println("订单已取消（已支付状态），退款处理完成");
                
            } else if (currentStatus == OrderStatus.SHIPPED) {
                // 已发货取消，需要拦截物流
                if (!interceptShipping(orderId)) {
                    System.out.println("物流拦截失败");
                    return false;
                }
                
                if (needRefund) {
                    boolean refundSuccess = processRefund(orderId, order.getAmount());
                    if (!refundSuccess) {
                        System.out.println("退款处理失败");
                        return false;
                    }
                    order.setRefunded(true);
                }
                
                order.setStatus(OrderStatus.CANCELLED);
                System.out.println("订单已取消（已发货状态），物流已拦截");
                
            } else {
                // 其他状态不能取消
                System.out.println("当前状态不能取消订单: " + currentStatus);
                return false;
            }
            
            // 记录取消原因
            logCancelReason(orderId, cancelReason);
            
            return true;
        }
        
        /**
         * 完成订单 - 更多的嵌套条件
         */
        public boolean completeOrder(String orderId) {
            Order order = orderStore.get(orderId);
            if (order == null) {
                System.out.println("订单不存在");
                return false;
            }
            
            // 检查当前状态
            if (order.getStatus() != OrderStatus.RECEIVED) {
                System.out.println("订单状态不是已收货，无法完成");
                return false;
            }
            
            // 检查是否已评价
            if (!isOrderReviewed(orderId)) {
                System.out.println("订单尚未评价，无法完成");
                return false;
            }
            
            // 检查自动确认时间是否已过
            if (!isAutoConfirmTimeout(orderId)) {
                System.out.println("自动确认时间未到，无法完成");
                return false;
            }
            
            // 更新状态
            order.setStatus(OrderStatus.COMPLETED);
            System.out.println("订单已完成");
            
            // 发放积分
            grantRewardPoints(orderId, (int) order.getAmount());
            
            return true;
        }
        
        /**
         * 申请售后 - 又一套条件判断
         */
        public boolean applyAfterSales(String orderId, String afterSalesType, String description) {
            Order order = orderStore.get(orderId);
            if (order == null) {
                System.out.println("订单不存在");
                return false;
            }
            
            // 检查当前状态
            if (order.getStatus() != OrderStatus.RECEIVED) {
                System.out.println("只有已收货订单可以申请售后");
                return false;
            }
            
            // 检查售后时效
            if (!isWithinAfterSalesPeriod(orderId)) {
                System.out.println("已超过售后申请时效");
                return false;
            }
            
            // 根据售后类型检查条件
            if ("RETURN".equals(afterSalesType)) {
                if (!isReturnAllowed(orderId)) {
                    System.out.println("该商品不支持退货");
                    return false;
                }
            } else if ("EXCHANGE".equals(afterSalesType)) {
                if (!isExchangeAllowed(orderId)) {
                    System.out.println("该商品不支持换货");
                    return false;
                }
            } else if ("REFUND".equals(afterSalesType)) {
                if (!isRefundAllowed(orderId)) {
                    System.out.println("该商品不支持退款");
                    return false;
                }
            } else {
                System.out.println("不支持的售后类型: " + afterSalesType);
                return false;
            }
            
            // 更新状态
            order.setStatus(OrderStatus.AFTER_SALES);
            System.out.println("售后申请已提交，状态变为售后处理");
            
            return true;
        }
        
        // 以下是各种辅助方法（模拟实现）
        private boolean checkInventory(String orderId) { return true; }
        private boolean processPayment(String orderId, double amount) { return true; }
        private void sendPaymentNotification(String orderId) {}
        private boolean validateShippingAddress(String orderId) { return true; }

        private void logCancelReason(String orderId, String cancelReason) {}
        private boolean isOrderReviewed(String orderId) { return true; }
        private boolean isAutoConfirmTimeout(String orderId) { return true; }
        private void grantRewardPoints(String orderId, int points) {}
        private boolean isWithinAfterSalesPeriod(String orderId) { return true; }
        private boolean isReturnAllowed(String orderId) { return true; }
        private boolean isExchangeAllowed(String orderId) { return true; }
        private boolean isRefundAllowed(String orderId) { return true; }
    }
```

无状态机版本的问题分析：

1. 代码重复：每个方法都有相同的空值检查和状态检查
2. 逻辑分散：状态转换规则分散在各个方法中
3. 难以扩展：新增状态或操作需要修改多个方法
4. 容易出错：条件判断复杂，容易遗漏边界情况
5. 难以测试：每个方法都有一堆前置条件

### 2.3 有状态机实现

```java
// OrderServiceWithStateMachine.java
import java.time.LocalDateTime;
import java.util.*;

/**
 * 使用状态机的订单服务实现
 * 特点：状态转换逻辑集中，代码清晰，易于扩展
 */
public class OrderServiceWithStateMachine {
    
    // 1. 定义状态枚举
    public enum OrderState {
        PENDING,       // 待支付
        PAID,          // 已支付
        SHIPPED,       // 已发货
        RECEIVED,      // 已收货
        COMPLETED,     // 已完成
        CANCELLED,     // 已取消
        AFTER_SALES    // 售后处理
        
        // 判断是否为终态
        public boolean isTerminal() {
            return this == COMPLETED || this == CANCELLED;
        }
    }
    
    // 2. 定义事件枚举
    public enum OrderEvent {
        PAY,           // 支付
        SHIP,          // 发货
        CONFIRM_RECEIPT, // 确认收货
        COMPLETE,      // 完成订单
        CANCEL,        // 取消订单
        APPLY_AFTER_SALES, // 申请售后
        PROCESS_REFUND,    // 处理退款
        REJECT_AFTER_SALES // 拒绝售后
    }
    
    // 3. 状态机上下文
    static class OrderStateContext {
        private String orderId;
        private OrderState currentState;
        private LocalDateTime stateTime;
        private Map<String, Object> contextData;
        
        public OrderStateContext(String orderId, OrderState initialState) {
            this.orderId = orderId;
            this.currentState = initialState;
            this.stateTime = LocalDateTime.now();
            this.contextData = new HashMap<>();
        }
        
        public void transitTo(OrderState newState) {
            System.out.printf("状态转换: %s -> %s%n", this.currentState, newState);
            this.currentState = newState;
            this.stateTime = LocalDateTime.now();
        }
        
        // 省略getter/setter...
    }
    
    // 4. 状态转移规则定义
    static class StateTransition {
        private OrderState fromState;
        private OrderEvent event;
        private OrderState toState;
        private String condition;  // 条件检查
        private String action;     // 执行动作
        
        public StateTransition(OrderState fromState, OrderEvent event, 
                              OrderState toState, String condition, String action) {
            this.fromState = fromState;
            this.event = event;
            this.toState = toState;
            this.condition = condition;
            this.action = action;
        }
        
        // 省略getter...
    }
    
    // 5. 状态机引擎
    static class OrderStateMachine {
        private Map<String, List<StateTransition>> transitionMap = new HashMap<>();
        private OrderStateContext context;
        private Order order;
        
        public OrderStateMachine(Order order) {
            this.order = order;
            this.context = new OrderStateContext(order.getOrderId(), order.getStatus());
            initializeTransitions();
        }
        
        /**
         * 初始化状态转移规则 - 所有规则集中在这里！
         */
        private void initializeTransitions() {
            // 格式：fromState + event -> (toState, condition, action)
            addTransition(OrderState.PENDING, OrderEvent.PAY, OrderState.PAID,
                "checkInventory && validatePayment", 
                "processPayment && sendPaymentNotification");
            
            addTransition(OrderState.PENDING, OrderEvent.CANCEL, OrderState.CANCELLED,
                null, 
                "logCancelReason");
            
            addTransition(OrderState.PAID, OrderEvent.SHIP, OrderState.SHIPPED,
                "validateShippingInfo", 
                "processShipping && sendShippingNotification");
            
            addTransition(OrderState.PAID, OrderEvent.CANCEL, OrderState.CANCELLED,
                null, 
                "processRefund && logCancelReason");
            
            addTransition(OrderState.SHIPPED, OrderEvent.CONFIRM_RECEIPT, OrderState.RECEIVED,
                "validateReceiptCode", 
                "startAutoConfirmTimer");
            
            addTransition(OrderState.SHIPPED, OrderEvent.CANCEL, OrderState.CANCELLED,
                "interceptShipping", 
                "processRefund && logCancelReason");
            
            addTransition(OrderState.RECEIVED, OrderEvent.COMPLETE, OrderState.COMPLETED,
                "isOrderReviewed && isAutoConfirmTimeout", 
                "grantRewardPoints");
            
            addTransition(OrderState.RECEIVED, OrderEvent.APPLY_AFTER_SALES, OrderState.AFTER_SALES,
                "isWithinAfterSalesPeriod && validateAfterSalesType", 
                "createAfterSalesRecord");
        }
        
        private void addTransition(OrderState from, OrderEvent event, OrderState to,
                                  String condition, String action) {
            String key = from.name() + "_" + event.name();
            transitionMap.computeIfAbsent(key, k -> new ArrayList<>())
                .add(new StateTransition(from, event, to, condition, action));
        }
        
        /**
         * 触发事件，执行状态转换
         */
        public boolean trigger(OrderEvent event, Map<String, Object> eventData) {
            String key = context.getCurrentState().name() + "_" + event.name();
            List<StateTransition> transitions = transitionMap.get(key);
            
            if (transitions == null || transitions.isEmpty()) {
                System.out.println("不支持的状态转换: " + context.getCurrentState() + " -> " + event);
                return false;
            }
            
            // 合并事件数据到上下文
            if (eventData != null) {
                context.getContextData().putAll(eventData);
            }
            
            // 找到匹配的转换规则
            for (StateTransition transition : transitions) {
                if (checkConditions(transition.getCondition())) {
                    executeActions(transition.getAction());
                    context.transitTo(transition.getToState());
                    order.setStatus(transition.getToState());
                    return true;
                }
            }
            
            System.out.println("条件检查未通过，状态转换失败");
            return false;
        }
        
        /**
         * 检查转换条件
         */
        private boolean checkConditions(String condition) {
            if (condition == null || condition.trim().isEmpty()) {
                return true;
            }
            
            // 这里可以解析条件表达式，简化实现
            String[] conditions = condition.split("&&");
            for (String cond : conditions) {
                cond = cond.trim();
                switch (cond) {
                    case "checkInventory":
                        if (!checkInventory()) return false;
                        break;
                    case "validatePayment":
                        if (!validatePayment()) return false;
                        break;
                    case "validateShippingInfo":
                        if (!validateShippingInfo()) return false;
                        break;
                    // ... 其他条件检查
                    default:
                        System.out.println("未知条件: " + cond);
                        return false;
                }
            }
            return true;
        }
        
        /**
         * 执行转换动作
         */
        private void executeActions(String action) {
            if (action == null || action.trim().isEmpty()) {
                return;
            }
            
            String[] actions = action.split("&&");
            for (String act : actions) {
                act = act.trim();
                switch (act) {
                    case "processPayment":
                        processPayment();
                        break;
                    case "sendPaymentNotification":
                        sendPaymentNotification();
                        break;
                    case "processShipping":
                        processShipping();
                        break;
                    // ... 其他动作
                }
            }
        }
```

## 06.状态机核心要素

### 6.1 状态机组成要素

状态机的几个核心要素：

1. 状态：系统所有可能的情况被精确定义为有限个数的“状态”（如状态A、状态B）。这是系统行为的“模式”。
2. 事件：来自外部或内部的“事件”（如事件X、事件Y)是触发状态迁移的输入信号。
3. 转移：定义在特定状态下，响应某个事件，并可能满足特定条件时，系统将执行一个动作并迁移到新的状态（或保持原状态）。
4. 动作：状态转移发生时可以伴随一个可执行的动作。

### 8.1 状态机基础框架

```cpp
#include <iostream>
#include <memory>
#include <unordered_map>
#include <functional>

// 前向声明
class StateMachine;
class State;

// 事件基类
class Event {
public:
    virtual ~Event() = default;
    virtual std::string getName() const = 0;
};

// 具体事件类
class StartEvent : public Event {
public:
    std::string getName() const override { return "StartEvent"; }
};

class StopEvent : public Event {
public:
    std::string getName() const override { return "StopEvent"; }
};

class PauseEvent : public Event {
public:
    std::string getName() const override { return "PauseEvent"; }
};

class ResumeEvent : public Event {
public:
    std::string getName() const override { return "ResumeEvent"; }
};

// 状态基类
class State {
public:
    virtual ~State() = default;
    virtual std::string getName() const = 0;
    virtual void onEnter(StateMachine* sm) {}
    virtual void onExit(StateMachine* sm) {}
    virtual std::shared_ptr<State> handleEvent(
        StateMachine* sm, 
        const Event& event
    ) = 0;
};

// 状态机类
class StateMachine {
private:
    std::shared_ptr<State> currentState_;
    std::unordered_map<std::string, std::shared_ptr<State>> states_;
    
public:
    StateMachine() = default;
    
    // 添加状态
    void addState(const std::string& name, std::shared_ptr<State> state) {
        states_[name] = state;
    }
    
    // 设置初始状态
    void setInitialState(const std::string& stateName) {
        auto it = states_.find(stateName);
        if (it != states_.end()) {
            currentState_ = it->second;
            currentState_->onEnter(this);
            std::cout << "状态机启动，初始状态: " << currentState_->getName() << std::endl;
        }
    }
    
    // 处理事件
    void handleEvent(const Event& event) {
        if (!currentState_) {
            std::cout << "错误: 状态机未初始化" << std::endl;
            return;
        }
        
        std::cout << "当前状态: " << currentState_->getName() 
                  << ", 接收事件: " << event.getName() << std::endl;
        
        auto nextState = currentState_->handleEvent(this, event);
        
        if (nextState && nextState != currentState_) {
            // 执行状态转换
            currentState_->onExit(this);
            std::cout << "状态转换: " << currentState_->getName() 
                      << " -> " << nextState->getName() << std::endl;
            currentState_ = nextState;
            currentState_->onEnter(this);
        }
    }
    
    // 获取当前状态
    std::string getCurrentStateName() const {
        return currentState_ ? currentState_->getName() : "None";
    }
    
    // 获取状态
    std::shared_ptr<State> getState(const std::string& name) {
        auto it = states_.find(name);
        return (it != states_.end()) ? it->second : nullptr;
    }
};
```

```cpp
class IdleState : public State {
public:
    std::string getName() const override { return "Idle"; }
    
    void onEnter(StateMachine* sm) override {
        std::cout << "进入空闲状态 - 系统准备就绪" << std::endl;
    }
    
    void onExit(StateMachine* sm) override {
        std::cout << "退出空闲状态" << std::endl;
    }
    
    std::shared_ptr<State> handleEvent(StateMachine* sm, const Event& event) override {
        if (event.getName() == "StartEvent") {
            return sm->getState("Running");
        }
        std::cout << "空闲状态忽略事件: " << event.getName() << std::endl;
        return nullptr;
    }
};

// 运行状态
class RunningState : public State {
public:
    std::string getName() const override { return "Running"; }
    
    void onEnter(StateMachine* sm) override {
        std::cout << "进入运行状态 - 开始执行任务" << std::endl;
    }
    
    void onExit(StateMachine* sm) override {
        std::cout << "退出运行状态 - 停止执行任务" << std::endl;
    }
    
    std::shared_ptr<State> handleEvent(StateMachine* sm, const Event& event) override {
        if (event.getName() == "PauseEvent") {
            return sm->getState("Paused");
        } else if (event.getName() == "StopEvent") {
            return sm->getState("Idle");
        }
        std::cout << "运行状态忽略事件: " << event.getName() << std::endl;
        return nullptr;
    }
};

    void onEnter(StateMachine* sm) override {
        std::cout << "进入暂停状态 - 任务已暂停" << std::endl;
    }
    
    void onExit(StateMachine* sm) override {
        std::cout << "退出暂停状态" << std::endl;
    }
    
    std::shared_ptr<State> handleEvent(StateMachine* sm, const Event& event) override {
        if (event.getName() == "ResumeEvent") {
            return sm->getState("Running");
        } else if (event.getName() == "StopEvent") {
            return sm->getState("Idle");
        }

        return nullptr;
```

## 10.总结和分析

### 10.1 核心要点回顾

1. **应用场景广泛**：从游戏开发到业务流程，从网络协议到用户界面
2. **解决复杂性问题**：将复杂的条件逻辑转换为清晰的状态转换
3. **设计思想明确**：状态驱动、事件响应、确定性转换
4. **实现相对简单**：基于面向对象的设计模式，易于理解和实现
5. **存在固有问题**：状态爆炸、性能开销、设计复杂度

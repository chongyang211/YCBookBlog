### 1.3 没有内存模型会发生什么

```mermaid
sequenceDiagram
    participant T1 as 线程1
    participant Cache1 as CPU1缓存
    participant Memory as 主内存
    participant Cache2 as CPU2缓存
    participant T2 as 线程2
    
    Note over T1,T2: 没有内存模型的混乱场景
    
    T1->>Cache1: 写入 x = 1
    Note over Cache1: 数据在缓存中
    
    T2->>Cache2: 读取 x
    Cache2->>Memory: 从主内存读取
    Memory-->>Cache2: 返回 x = 0 (旧值)
    Cache2-->>T2: x = 0
    
    Note over T1,T2: 线程2看不到线程1的修改！
    
    T1->>Cache1: 写入 y = 2
    T1->>Memory: 延迟写回主内存
    
    T2->>Cache2: 读取 y
    Cache2->>Memory: 从主内存读取
    Memory-->>Cache2: 返回 y = 0 (还未更新)
    
    Note over T1,T2: 数据不一致，程序行为不可预测
```

### 2.2 线程私有 vs 线程共享对比

| 特性 | 线程私有内存 | 线程共享内存 |
|------|-------------|-------------|
| **访问权限** | 仅当前线程可访问 | 所有线程都可访问 |
| **生命周期** | 与线程同生共死 | 与进程同生共死 |
| **数据安全** | 天然线程安全 | 需要同步机制 |
| **性能** | 访问速度快 | 可能存在竞争 |
| **内存大小** | 相对较小 | 相对较大 |
| **典型用途** | 方法调用、局部变量 | 对象实例、类信息 |

```java
// 栈内存使用示例
public class StackExample {
    public void methodA() {
        int localVar = 10;    // 存储在栈帧的局部变量表
        String str = "Hello"; // 引用存储在栈，对象在堆

        for (int i = 0; i < 5; i++) {
            array[i] = i * param; // 操作数栈参与计算
        }
    } // 栈帧销毁
}
```

#### 3.3.2 实际场景案例

```java
// JNI本地方法调用示例
public class NativeExample {
    // 声明本地方法
    public native int calculateHash(String input);
    public native void processImage(byte[] imageData);
    
    static {
        // 加载本地库
        System.loadLibrary("nativelib");
    }
    
    public void useNativeMethod() {
        String data = "Hello World";
        // 调用本地方法，使用本地方法栈
        int hash = calculateHash(data);
        System.out.println("Hash: " + hash);
    }
}
```

```c
// C语言实现的本地方法
#include <jni.h>
#include <string.h>

JNIEXPORT jint JNICALL 
Java_NativeExample_calculateHash(JNIEnv *env, jobject obj, jstring input) {
    // 本地方法栈中的局部变量
    const char *str = (*env)->GetStringUTFChars(env, input, 0);
    int hash = 0;
    
    // 在本地方法栈中执行计算
    for (int i = 0; i < strlen(str); i++) {
        hash = hash * 31 + str[i];
    }
    
    (*env)->ReleaseStringUTFChars(env, input, str);
    return hash;
}
```

#### 4.1.2 实际场景案例

```java
// 堆内存使用示例
public class HeapExample {
    private static List<String> staticList = new ArrayList<>(); // 在堆中
    
    public void createObjects() {
        // 所有对象都在堆内存中创建
        String str = new String("Hello");     // 在堆中
        StringBuilder sb = new StringBuilder(); // 在堆中
        int[] array = new int[1000];          // 在堆中
        
        // 对象引用在栈中，对象实例在堆中
        Person person = new Person("张三", 25);
        
        // 集合对象在堆中，元素也在堆中
        List<Person> persons = new ArrayList<>();
        persons.add(person);
        
        // 大对象可能直接进入老年代
        byte[] bigArray = new byte[1024 * 1024]; // 1MB
    }
    
    static class Person {
        private String name;  // 引用在堆中，字符串对象也在堆中
        private int age;      // 基本类型值在堆中（作为对象的字段）
        
        public Person(String name, int age) {
            this.name = name;
            this.age = age;
        }
    }
}
```

#### 4.1.3 垃圾回收过程

```mermaid
sequenceDiagram
    participant App as 应用程序
    participant Eden as Eden区
    participant S0 as Survivor0
    participant S1 as Survivor1
    participant Old as 老年代
    participant GC as 垃圾回收器
    
    App->>Eden: 创建新对象
    Eden->>Eden: Eden区满了

    Eden->>GC: 再次触发Minor GC
    GC->>S1: S0+Eden存活对象移到S1
    GC->>S0: 清空S0
    
    Note over S1: 对象年龄+1
    
    S1->>Old: 年龄达到阈值，晋升到老年代
    
    Old->>GC: 老年代满了，触发Major GC
```

#### 4.4.1 直接内存特点

#### 4.4.2 实际场景案例

```java
// 直接内存使用示例
import java.nio.ByteBuffer;
import java.nio.channels.FileChannel;
import java.io.RandomAccessFile;

public class DirectMemoryExample {
    
    public void directBufferExample() {
        // 分配直接内存
        ByteBuffer directBuffer = ByteBuffer.allocateDirect(1024 * 1024); // 1MB
        ByteBuffer heapBuffer = ByteBuffer.allocate(1024 * 1024);         // 堆内存
        
        // 直接内存的优势：零拷贝
        byte[] data = "Hello Direct Memory".getBytes();
        directBuffer.put(data);
        directBuffer.flip();
        
        // 读取数据
        byte[] result = new byte[data.length];
        directBuffer.get(result);
        
        System.out.println(new String(result));
    }
    
    public void fileChannelExample() throws Exception {
        RandomAccessFile file = new RandomAccessFile("test.txt", "rw");
        FileChannel channel = file.getChannel();
        
        // 使用直接内存进行文件操作
        ByteBuffer buffer = ByteBuffer.allocateDirect(1024);
        
        // 零拷贝读取文件
        int bytesRead = channel.read(buffer);
        buffer.flip();
        
        // 零拷贝写入文件
        channel.write(buffer);
        
        channel.close();
        file.close();
    }
    
    // 内存映射文件示例
    public void memoryMappedFileExample() throws Exception {
        RandomAccessFile file = new RandomAccessFile("large_file.dat", "rw");
        FileChannel channel = file.getChannel();
        
        // 内存映射，使用直接内存
        long fileSize = 1024 * 1024 * 100; // 100MB
        var mappedBuffer = channel.map(
            FileChannel.MapMode.READ_WRITE, 0, fileSize
        );
        
        // 直接操作内存映射区域
        mappedBuffer.put(0, (byte) 'A');
        byte value = mappedBuffer.get(0);
        
        channel.close();
        file.close();
    }
}
```

```java
// 栈内存工作原理示例
public class StackPrincipleExample {
    
    public void demonstrateStackOperation() {
        // 栈帧1开始
        int a = 10;        // 局部变量表slot 0
        double b = 3.14;   // 局部变量表slot 1-2 (double占2个slot)
        
        String str = processData(a, b); // 方法调用，创建新栈帧
        
        System.out.println(str);
        // 栈帧1结束，所有局部变量自动销毁
    }
    
    private String processData(int num, double decimal) {
        // 栈帧2开始
        // 参数：num在slot 0, decimal在slot 1-2
        StringBuilder sb = new StringBuilder(); // 局部变量slot 3
        
        // 操作数栈操作
        sb.append("Number: ");  // sb引用压入操作数栈
        sb.append(num);         // num值压入操作数栈，调用append
        sb.append(", Decimal: ");
        sb.append(decimal);     // decimal值压入操作数栈
        
        String result = sb.toString(); // 调用toString，结果压入操作数栈
        return result;          // 返回值通过操作数栈传递
        // 栈帧2结束，局部变量自动销毁
    }
}
```

```java
/**
 * 综合内存模型案例：在线购物系统
 * 演示各种内存区域的使用
 */
public class OnlineShoppingSystem {
    
    // 静态变量：引用在方法区，对象在堆中
    private static final String SYSTEM_NAME = "购物系统"; // 字符串常量池
    private static Map<String, Product> productCache = new ConcurrentHashMap<>(); // 堆中
    
    // 实例变量：存储在堆中（作为对象的一部分）
    private String userId;
    private List<CartItem> shoppingCart;
    
    public OnlineShoppingSystem(String userId) {
        this.userId = userId;                    // 字符串在堆中
        this.shoppingCart = new ArrayList<>();   // 集合对象在堆中
    }
    
    /**
     * 添加商品到购物车
     * 演示栈内存、堆内存、方法区的协作
     */
    public void addToCart(String productId, int quantity) {
        // 局部变量在栈中
        long startTime = System.currentTimeMillis(); // 栈中
        
        try {
            // 从缓存获取商品（堆内存访问）
            Product product = getProductFromCache(productId);
            
            if (product == null) {
                // 从数据库加载（可能使用直接内存进行IO）
                product = loadProductFromDatabase(productId);
                
                if (product != null) {
                    // 加入缓存（堆内存操作）
                    productCache.put(productId, product);
                }
            }
            
            if (product != null && product.getStock() >= quantity) {
                // 创建购物车项（堆内存分配）
                CartItem item = new CartItem(product, quantity);
                
                // 添加到购物车（堆内存中的集合操作）
                shoppingCart.add(item);
                
                // 更新库存（堆内存中对象状态修改）
                product.reduceStock(quantity);
                
                // 记录日志（可能使用直接内存进行文件IO）
                logOperation("添加商品", productId, quantity);
            }
            
        } catch (Exception e) {
            // 异常对象在堆中创建
            handleException(e);
        } finally {
            // 计算耗时（栈中局部变量）
            long duration = System.currentTimeMillis() - startTime;
            System.out.println("操作耗时: " + duration + "ms");
        }
    }
    
    /**
     * 使用本地方法进行性能优化
     */
    public native double calculateOptimalPrice(double basePrice, int quantity);
    
    /**
     * 异步处理订单（演示线程私有内存）
     */
    public CompletableFuture<Order> processOrderAsync() {
        return CompletableFuture.supplyAsync(() -> {
            // 新线程有自己的栈内存
            String threadName = Thread.currentThread().getName(); // 栈中
            long threadId = Thread.currentThread().getId();       // 栈中
            
            try {
                // 创建订单对象（堆内存）
                Order order = new Order(userId, new ArrayList<>(shoppingCart));
                
                // 计算总价（栈中临时变量）
                double totalPrice = 0.0;
                for (CartItem item : shoppingCart) {
                    double itemPrice = item.getProduct().getPrice() * item.getQuantity();
                    
                    // 使用本地方法优化价格计算（本地方法栈）
                    itemPrice = calculateOptimalPrice(itemPrice, item.getQuantity());
                    
                    totalPrice += itemPrice;
                }
                
                order.setTotalPrice(totalPrice);
                
                // 保存订单（可能使用直接内存进行数据库IO）
                saveOrderToDatabase(order);
                
                return order;
                
            } catch (Exception e) {
                throw new RuntimeException("订单处理失败", e);
            }
        });
    }
    
    // 辅助方法
    private Product getProductFromCache(String productId) {
        return productCache.get(productId); // 访问堆中的缓存
    }
    
    private Product loadProductFromDatabase(String productId) {
        // 模拟数据库查询，可能使用直接内存进行IO
        // 实际实现会使用JDBC等
        return new Product(productId, "商品名称", 99.99, 100);
    }
    
    private void logOperation(String operation, String productId, int quantity) {
        // 日志记录，可能使用直接内存写入文件
        String logMessage = String.format("[%s] %s: 商品=%s, 数量=%d", 
            new Date(), operation, productId, quantity);
        System.out.println(logMessage);
    }
    
    private void handleException(Exception e) {
        // 异常处理逻辑
        e.printStackTrace();
    }
    
    private void saveOrderToDatabase(Order order) {
        // 模拟数据库保存操作
        System.out.println("订单已保存: " + order.getOrderId());
    }
}
```

```go
// Goroutine栈管理
func demonstrateGoroutineStack() {
    var wg sync.WaitGroup
    
    for i := 0; i < 10; i++ {
        wg.Add(1)
        go func(id int) {
            defer wg.Done()
            
            // 每个goroutine有自己的栈
            var localArray [1000]int
            localArray[0] = id
            
            // 递归调用会增长栈
            recursiveFunction(10, id)
            
            fmt.Printf("Goroutine %d 完成\n", id)
        }(i)
    }
    
    wg.Wait()
}

func recursiveFunction(depth, id int) {
    if depth <= 0 {
        return
    }
    
    // 栈帧增长
    var localVar [100]int
    localVar[0] = depth
    
    recursiveFunction(depth-1, id)
}
```

```rust
fn demonstrate_thread_safety() {

    }

}
```

```java
class ObjectPool<T> {
    private final Queue<T> pool = new ConcurrentLinkedQueue<>();
    private final Supplier<T> factory;
    private final int maxSize;
    
    public ObjectPool(Supplier<T> factory, int maxSize) {
        this.factory = factory;
        this.maxSize = maxSize;
    }
    
    public T borrow() {
        T obj = pool.poll();
        return obj != null ? obj : factory.get();
    }
    
    public void returnObject(T obj) {
        if (pool.size() < maxSize) {
            pool.offer(obj);
        }
    }
```
